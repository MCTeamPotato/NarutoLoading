package me.kall.narutoloading.agent.bootstrap;

import java.util.function.Function;

public final class EarlyBridge {
    public static final String ACTIVE_PROPERTY = "narutoloading.early";
    public static final String SESSION_PROPERTY = "narutoloading.early.session";
    public static final String MEDIA_PROPERTY = "narutoloading.early.media";

    private static volatile Function<ClassLoader, EarlyVideo> factory;
    private static volatile EarlyVideo video;
    private static volatile boolean created;

    private EarlyBridge() {
    }

    public static void install(Function<ClassLoader, EarlyVideo> earlyFactory) {
        factory = earlyFactory;
        System.setProperty(ACTIVE_PROPERTY, "1");
    }

    public static void render(ClassLoader loader) {
        EarlyVideo current = video;

        if (current == null) {
            if (created) return;
            created = true;

            Function<ClassLoader, EarlyVideo> earlyFactory = factory;
            if (earlyFactory == null) return;

            try {
                current = earlyFactory.apply(loader);
            } catch (Throwable throwable) {
                System.err.println("[NarutoLoading Agent] Failed to create the early video: " + throwable.getMessage());
                return;
            }

            video = current;
            if (current == null) return;
        }

        current.present(true);
    }

    @SuppressWarnings("unused")
    public static void detach() {
        factory = null;
        System.clearProperty(ACTIVE_PROPERTY);

        EarlyVideo current = video;
        video = null;

        if (current != null) System.getProperties().put(SESSION_PROPERTY, current);
    }
}

    private static volatile Method patcher;

    @Override
    public byte @Nullable [] transform(ClassLoader loader, String className, Class<?> classBeingRedefined, ProtectionDomain protectionDomain, byte[] classfileBuffer) {
        if (!DISPLAY_WINDOW.equals(className)) return null;

        try {
            Method patch = patcher();

        } catch (Throwable throwable) {

            return null;
        }
    }

    private static @Nullable Method patcher() {
        Method patch = patcher;
        if (patch != null) return patch;

        synchronized (DisplayWindowTransformer.class) {
            patch = patcher;

            if (patch == null) {

                    patch = Class.forName("me.kall.narutoloading.agent.DisplayWindowPatcher", true, DisplayWindowTransformer.class.getClassLoader()).getMethod("patch", byte[].class);

                } catch (ClassNotFoundException | NoSuchMethodException exception) {
                    System.err.println("[NarutoLoading Agent] Patch entry point missing: " + exception.getMessage());
                    return null;
                }
            }
        }

        return patch;
    }

package me.kall.narutoloading.agent;

import org.jetbrains.annotations.NotNull;

import java.net.URL;
import java.net.URLClassLoader;

final class EarlyClassLoader extends URLClassLoader {
    private static final String MOD_PACKAGE = "me.kall.narutoloading.";
    private static final String BOOTSTRAP_PACKAGE = "me.kall.narutoloading.agent.bootstrap.";

    EarlyClassLoader(URL[] urls, ClassLoader parent) {
        super(urls, parent);
    }

    @Override
    protected Class<?> loadClass(String name, boolean resolve) throws ClassNotFoundException {
        if (!isOwn(name)) return super.loadClass(name, resolve);

        synchronized (this.getClassLoadingLock(name)) {
            Class<?> loaded = this.findLoadedClass(name);

            if (loaded == null) {
                try {
                    loaded = this.findClass(name);
                } catch (ClassNotFoundException exception) {
                    loaded = super.loadClass(name, false);
                }
            }

            if (resolve) this.resolveClass(loaded);
            return loaded;
        }
    }

    private static boolean isOwn(@NotNull String name) {
        if (name.startsWith(BOOTSTRAP_PACKAGE)) return false;
        return name.startsWith(MOD_PACKAGE);
    }
}

import me.kall.narutoloading.config.NarutoConfig;
import org.jetbrains.annotations.Nullable;

import java.nio.file.Path;

final class EarlyMedia {
    final String ffmpeg;
    final VideoArgs video;
    final AudioArgs audio;
    final float volume;
    final double bufferSeconds;

    private EarlyMedia(String ffmpeg, VideoArgs video, AudioArgs audio, float volume, double bufferSeconds) {
        this.ffmpeg = ffmpeg;
        this.video = video;
        this.audio = audio;
        this.volume = volume;
        this.bufferSeconds = bufferSeconds;
    }

    static @Nullable EarlyMedia resolve() {
        Path gameDir = GamePaths.gameDir();
        String configuredDir = NarutoConfig.ffmpegDir().toString();

        String ffmpeg = Locator.locate("ffmpeg", gameDir, configuredDir);
        String ffprobe = Locator.locate("ffprobe", gameDir, configuredDir);
        if (ffmpeg == null || ffprobe == null) return null;

        MediaSource source = MediaRepository.pickSource(gameDir.resolve(NarutoConfig.sourcesDir()), NarutoConfig.selectMode(), NarutoConfig.fixedName());
        if (source == null) return null;

        FFmpeg tools = new FFmpeg();
        tools.ffmpeg.set(ffmpeg);
        tools.ffprobe.set(ffprobe);

        VideoArgs video = tools.deriveVideo(source.video());
        if (video == null) return null;

        AudioArgs audio = tools.deriveAudio(source.audio() != null ? source.audio() : source.video());

        return new EarlyMedia(ffmpeg, video, audio, NarutoConfig.volume(), Math.max(1D, NarutoConfig.bufferSeconds()));
    }
}

package me.kall.narutoloading.agent;

import me.kall.narutoloading.agent.bootstrap.EarlyVideo;
import me.kall.narutoloading.media.ffmpeg.data.AudioArgs;
import me.kall.narutoloading.media.ffmpeg.data.VideoArgs;
import me.kall.narutoloading.playback.render.BufferFrameRenderer;
import me.kall.narutoloading.config.NarutoConfig;
import me.kall.narutoloading.playback.produce.audio.AudioProducer;
import me.kall.narutoloading.playback.produce.video.AbstractFrameProducer;
import me.kall.narutoloading.playback.produce.video.BufferFrameProducer;
import org.jetbrains.annotations.NotNull;

import java.nio.ByteBuffer;
import java.util.concurrent.atomic.AtomicBoolean;

public final class EarlyVideoImpl implements EarlyVideo {
    private final VideoArgs video;
    private final BufferFrameProducer producer;
    private final BufferFrameRenderer renderer = new BufferFrameRenderer(false);
    private final AudioProducer audio;

    private final AtomicBoolean clockStarted = new AtomicBoolean();

    private volatile long startNanos = -1L;
    private volatile boolean closed;
    private long lastFrameIndex = -1L;

    private EarlyVideoImpl(String ffmpeg, VideoArgs video, AudioArgs audioArgs, float volume, double bufferSeconds) {
        this.video = video;

        AudioProducer created = null;

        if (audioArgs != null && volume > 0F) {
            try {
                created = new AudioProducer(ffmpeg, volume, audioArgs, bufferSeconds);
                created.setup(0D);
            } catch (Throwable throwable) {
                System.err.println("[NarutoLoading Agent] Early audio disabled: " + throwable.getMessage());
                created = null;
            }
        }

        this.audio = created;

        this.producer = new BufferFrameProducer(ffmpeg, video, false, bufferSeconds);
        this.producer.withWarmListener(this::onWarm);
        this.producer.setup(0D);
        this.producer.eager();

        this.renderer.setup(video.scaledWidth(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), video.scaledHeight(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), 0D, false);
    }

    public static @NotNull EarlyVideo create(String ffmpeg, String videoPath, int width, int height, double fps, double duration, double startSec, String audioPath, int channels, int sampleRate, double audioStartSec, double audioDuration, float volume, double bufferSeconds) {
        VideoArgs video = new VideoArgs(videoPath, fps, width, height, duration, startSec);
        AudioArgs audio = audioPath != null && channels > 0 && sampleRate > 0 ? new AudioArgs(audioPath, channels, sampleRate, audioStartSec, audioDuration) : null;

        return new EarlyVideoImpl(ffmpeg, video, audio, volume, bufferSeconds);
    }

    private void onWarm() {
        if (!this.clockStarted.compareAndSet(false, true)) return;

        this.startNanos = System.nanoTime();
        if (this.audio != null) this.audio.eager();
    }

    @Override
    public void present(boolean flipY) {
        if (this.closed) return;

        try {
            double position = this.positionSec();
            long frameIndex = (long) (position * this.video.fps());

            if (frameIndex != this.lastFrameIndex) {
                this.lastFrameIndex = frameIndex;

                AbstractFrameProducer.Frame<ByteBuffer> frame = this.producer.fetch(position);
                if (frame != null) this.renderer.update(this.video.scaledWidth(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), this.video.scaledHeight(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), frame.data());
            }

            this.renderer.render(flipY);
        } catch (Throwable throwable) {
            System.err.println("[NarutoLoading Agent] Early video disabled: " + throwable.getMessage());
            this.close();
        }
    }

    @Override
    public double positionSec() {
        long start = this.startNanos;
        return start < 0L ? 0D : (System.nanoTime() - start) / 1_000_000_000D;
    }

    @Override
    public boolean finished() {
        return this.closed || this.positionSec() >= this.video.duration();
    }

    @Override
    public void close() {
        if (this.closed) return;
        this.closed = true;

        if (this.audio != null) this.audio.shutdown();
        this.producer.shutdown();
        this.renderer.shutdown();
    }
}

package me.kall.narutoloading.agent;

import org.jetbrains.annotations.NotNull;

import java.io.IOException;
import java.io.InputStream;
import java.lang.instrument.Instrumentation;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.jar.JarFile;
import java.util.zip.ZipEntry;
import java.util.zip.ZipOutputStream;

public final class NarutoLoadingAgent {
    private static final String[] BOOTSTRAP_CLASSES = {"me/kall/narutoloading/agent/bootstrap/EarlyBridge.class", "me/kall/narutoloading/agent/bootstrap/EarlyVideo.class"};

    private NarutoLoadingAgent() {
    }

    public static void premain(String args, Instrumentation instrumentation) {
        try {
            instrumentation.appendToBootstrapClassLoaderSearch(new JarFile(createBootstrapJar().toFile()));
        } catch (Throwable throwable) {
            System.err.println("[NarutoLoading Agent] Failed to install the bootstrap classes: " + throwable.getMessage());
            return;
        }

        instrumentation.addTransformer(new DisplayWindowTransformer(), false);
        EarlyInstaller.install();
    }

    private static @NotNull Path createBootstrapJar() throws IOException {
        Path jar = Files.createTempFile("narutoloading-bootstrap", ".jar");
        jar.toFile().deleteOnExit();

        ClassLoader loader = NarutoLoadingAgent.class.getClassLoader();

        try (ZipOutputStream output = new ZipOutputStream(Files.newOutputStream(jar))) {
            for (String entry : BOOTSTRAP_CLASSES) {
                try (InputStream input = loader.getResourceAsStream(entry)) {
                    if (input == null) throw new IOException("Missing " + entry);

                    output.putNextEntry(new ZipEntry(entry));
                    input.transferTo(output);
                    output.closeEntry();
                }
            }
        }

        return jar;
    }
}

        PlaybackSession session = EarlySessionHandoff.current();

        if (session != null) {
            session.present(false);
            return;
        }

        active().tick();
    }

    public static void applyMode() {
        EarlySessionHandoff.stop();
        NarutoContext.get().modes().apply();
    }

    public static void toggleCompatMode() {
        NarutoConfig.writeAll(Map.of(NarutoSettings.COMPAT_MODE, !NarutoConfig.compatMode()));
        NarutoConfig.save();
        applyMode();
    }

    public static void restart(double seekTo) {
        EarlySessionHandoff.stop();
        active().synchronize(true, seekTo);
    }

    /**
     * Returns the playback position of the active session, or {@code 0} when nothing is playing.
     * Use it to restart the video at the point the player was watching instead of rewinding it.
     *
     * @return the current position in seconds
     */
    public static double positionSec() {
        TimeLine timeLine = active().timeLine.get();
        return timeLine == null ? 0D : timeLine.positionSec();
    }

    public static void detachEarly() {
        EarlySessionHandoff.detach();
    }
}

package me.kall.narutoloading.client.fade;

import me.kall.narutoloading.NarutoLoading;
import me.kall.narutoloading.config.NarutoConfig;
import me.kall.narutoloading.client.ClientUiControl;
import me.kall.narutoloading.client.context.LevelStatus;
import net.minecraft.client.Minecraft;
import net.minecraft.client.MouseHandler;
import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.client.event.InputEvent;
import net.minecraftforge.event.TickEvent;
import net.minecraftforge.eventbus.api.SubscribeEvent;
import net.minecraftforge.fml.common.Mod;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.util.function.BooleanSupplier;
import java.util.function.IntSupplier;

@Mod.EventBusSubscriber(value = Dist.CLIENT, modid = NarutoLoading.MOD_ID)
public final class FadeCenter {
    private static final MousePos LAST_MOUSE = new MousePos();

    private static volatile @Nullable BooleanSupplier previewEnabled;
    private static volatile @Nullable IntSupplier previewTicks;
    private static volatile @Nullable IntSupplier previewFadeOutStep;
    private static volatile @Nullable IntSupplier previewFadeInStep;

    private static int stopTicks = 0;
    private static int fadeAlpha = 100;

    public static void preview(@NotNull BooleanSupplier enabled, @NotNull IntSupplier ticksBeforeFade, @NotNull IntSupplier fadeOutStep, @NotNull IntSupplier fadeInStep) {
        previewEnabled = enabled;
        previewTicks = ticksBeforeFade;
        previewFadeOutStep = fadeOutStep;
        previewFadeInStep = fadeInStep;
    }

    public static void clearPreview() {
        previewEnabled = null;
        previewTicks = null;
        previewFadeOutStep = null;
        previewFadeInStep = null;
    }

    public static boolean enabled() {
        BooleanSupplier preview = previewEnabled;
        return preview == null ? NarutoConfig.fadable() : preview.getAsBoolean();
    }

    public static int ticksBeforeFade() {
        IntSupplier preview = previewTicks;
        return preview == null ? NarutoConfig.ticksBeforeFade() : preview.getAsInt();
    }

    public static int fadeOutStep() {
        IntSupplier preview = previewFadeOutStep;
        return preview == null ? NarutoConfig.fadeOutStep() : preview.getAsInt();
    }

    public static int fadeInStep() {
        IntSupplier preview = previewFadeInStep;
        return preview == null ? NarutoConfig.fadeInStep() : preview.getAsInt();
    }

    public static boolean isHidden() {
        return LevelStatus.notFound() && fadeAlpha < 3;
    }

    public static boolean isFull() {
        return !LevelStatus.notFound() || fadeAlpha == 100;
    }

    public static float fadeAlpha() {
        if (!LevelStatus.notFound()) return 1.0F;
        return (float) fadeAlpha / 100.0F;
    }

    public static int modifyAlpha(int color) {
        if (isFull()) return color;
        int alpha = (int) (((color >>> 24) & 0xFF) * fadeAlpha()) & 0xFF;
        return (color & 0x00FFFFFF) | (alpha << 24);
    }

    public static int modifyTextAlpha(int color) {
        if (isFull()) return color;
        int alpha = (color >>> 24) & 0xFF;
        if (alpha == 0) alpha = 0xFF;
        alpha = (int) (alpha * fadeAlpha()) & 0xFF;
        return (color & 0x00FFFFFF) | (alpha << 24);
    }

    @SubscribeEvent
    public static void tick(TickEvent.@NotNull ClientTickEvent event) {
        if (!event.phase.equals(TickEvent.Phase.START)) return;

        if (!LevelStatus.notFound() || !ClientUiControl.videoPresent() || !enabled()) {
            fadeAlpha = 100;
            stopTicks = 0;
            LAST_MOUSE.set(Double.NaN, Double.NaN);
            return;
        }

        Minecraft minecraft = Minecraft.getInstance();
        MouseHandler mouseHandler = minecraft.mouseHandler;

        double x = mouseHandler.xpos();
        double y = mouseHandler.ypos();

        if (minecraft.screen != null && minecraft.getOverlay() == null && !LAST_MOUSE.init(x, y)) {
            stopTicks = LAST_MOUSE.same(x, y) ? stopTicks + 1 : 0;
            LAST_MOUSE.set(x, y);
        } else {
            stopTicks = 0;
        }

        fadeAlpha = stopTicks >= ticksBeforeFade() ? fadeAlpha - Math.max(0, fadeOutStep()) : fadeAlpha + Math.max(0, fadeInStep());
        if (fadeAlpha < 0) fadeAlpha = 0;
        if (fadeAlpha > 100) fadeAlpha = 100;
    }

    @SubscribeEvent
    public static void anyInput(InputEvent event) {
        reset();
    }

    @SubscribeEvent
    public static void mouseInput(InputEvent.MouseButton.Pre event) {
        reset();
    }

    private static void reset() {
        Minecraft.getInstance().execute(() -> {
            fadeAlpha = 100;
            stopTicks = 0;
            LAST_MOUSE.set(Double.NaN, Double.NaN);
        });
    }

    private static final class MousePos {
        double x = Double.NaN;
        double y = Double.NaN;

        boolean same(double x, double y) {
            return this.x == x && this.y == y;
        }

        boolean init(double x, double y) {
            if (Double.isNaN(this.x) || Double.isNaN(this.y)) {
                this.x = x;
                this.y = y;
                return true;
            }

            return false;
        }

        void set(double x, double y) {
            this.x = x;
            this.y = y;
        }
    }
}

package me.kall.narutoloading.client.ui;

import me.kall.narutoloading.config.NarutoConfig;
import me.kall.narutoloading.config.NarutoSettings;
import me.kall.narutoloading.config.Setting;
import org.jetbrains.annotations.NotNull;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;

final class NarutoDraft {
    private final Map<Setting<?>, Object> saved;
    private final Map<Setting<?>, Object> values;

    private NarutoDraft(@NotNull Map<Setting<?>, Object> saved) {
        this.saved = new LinkedHashMap<>(saved);
        this.values = new LinkedHashMap<>(saved);
    }

    static @NotNull NarutoDraft load() {
        return new NarutoDraft(NarutoConfig.snapshot());
    }

    @SuppressWarnings("unchecked")
    <T> T get(@NotNull Setting<T> setting) {
        return (T) this.values.get(setting);
    }

    <T> void set(@NotNull Setting<T> setting, T value) {
        this.values.put(setting, value);
    }

    boolean dirty() {
        for (Setting<?> setting : NarutoSettings.ALL) {
            if (this.changed(setting)) return true;
        }

        return false;
    }

    boolean playbackChanged() {
        for (Setting<?> setting : NarutoSettings.ALL) {
            if (!setting.playback()) continue;
            if (this.changed(setting)) return true;
        }

        return false;
    }

    void write() {
        Map<Setting<?>, Object> changed = new LinkedHashMap<>();

        for (Setting<?> setting : NarutoSettings.ALL) {
            if (this.changed(setting)) changed.put(setting, this.values.get(setting));
        }

        NarutoConfig.writeAll(changed);
        NarutoConfig.save();

        this.saved.clear();
        this.saved.putAll(this.values);
    }

    private boolean changed(@NotNull Setting<?> setting) {
        return !Objects.equals(this.values.get(setting), this.saved.get(setting));
    }
}

    private @NotNull Screen build(@Nullable Screen parent) {
        FadeCenter.preview(() -> this.draft.get(NarutoSettings.FADABLE), () -> this.draft.get(NarutoSettings.TICKS_BEFORE_FADE), () -> this.draft.get(NarutoSettings.FADE_OUT_STEP), () -> this.draft.get(NarutoSettings.FADE_IN_STEP));

        return MuiSettingsScreen.builder("NarutoLoading")
                .subtitle("narutoloading.screen.subtitle")
                .widgetFactory(MuiWidgetFactory.shimmer())
                .shimmerProfile(this.shimmerProfile)
                .shimmerNavigation()
                .shimmerTitles()
                .railHeader((context, theme) -> MuiLogo.create(context, theme, "N"))
                .railFooter((context, theme) -> MuiHintBadge.create(context, theme, "narutoloading.ytdlp.footer"))
                .actionBar(this::buildActionBar)
                .onCreated(fragment -> {
                    this.host = fragment;
                    this.refreshActions();
                })
                .onDestroyed(fragment -> this.onDestroyed())
                .onCloseRequested(this::onCloseRequested)
                .page("narutoloading.nav.download", "narutoloading.ytdlp.download.title", "narutoloading.ytdlp.download.subtitle", this.downloads::build)
                .page("narutoloading.nav.playback", "narutoloading.playback.title", "narutoloading.playback.subtitle", this.playback::build)
                .page("narutoloading.nav.config", "narutoloading.config.title", "narutoloading.config.subtitle", this.config::build)
                .page("narutoloading.nav.help", "narutoloading.help.title", "narutoloading.help.subtitle", HelpPage::build)
                .create(parent);
    }

    @NotNull NarutoDraft draft() {
        return this.draft;
    }

    void edit(@NotNull Runnable change) {
        change.run();
        this.refreshActions();
    }

    void reloadPage() {
        MuiScreenHost current = this.host;
        if (current != null) current.reloadCurrentPage();
    }

    void notifyImport(@NotNull String message) {
        MuiModApi.postToUiThread(() -> {
            MuiScreenHost current = this.host;
            if (current != null) MuiDialog.notice(current, I18n.get("narutoloading.playback.import.title"), message, I18n.get("narutoloading.ok"), () -> {});
        });
    }

    void applyVolume() {
        AudioProducer audio = ClientUiControl.active().audio.get();
        if (audio != null) audio.volume(this.draft.get(NarutoSettings.AUDIO_VOLUME));
    }

    private @NotNull View buildActionBar(@NotNull Context context) {
        MuiTheme theme = MuiTheme.modern(context);
        LinearLayout bar = MuiControls.row(context);
        bar.setGravity(Gravity.CENTER_VERTICAL);
        bar.addView(new View(context), new LinearLayout.LayoutParams(0, 1, 1.0F));

        this.statusView = MuiControls.text(context, theme, "", 12, theme.onSurfaceVariant());
        LinearLayout.LayoutParams statusParams = MuiControls.wrap();
        statusParams.rightMargin = MuiControls.dp(context, 8);
        bar.addView(this.statusView, statusParams);

        this.revertButton = MuiControls.button(context, theme, "narutoloading.action.revert", true, view -> this.revert(), MuiShimmerable.ALWAYS);
        this.addButton(context, bar, this.revertButton);

        this.applyButton = MuiControls.button(context, theme, "narutoloading.action.apply", true, view -> this.apply(false), MuiShimmerable.ALWAYS);
        this.addButton(context, bar, this.applyButton);

        this.doneButton = MuiControls.button(context, theme, "narutoloading.action.done", false, view -> this.finish(), MuiShimmerable.ALWAYS);
        this.addButton(context, bar, this.doneButton);

        this.refreshActions();
        return bar;
    }

    private void addButton(@NotNull Context context, @NotNull LinearLayout bar, @NotNull Button button) {
        LinearLayout.LayoutParams params = MuiControls.wrap();
        params.leftMargin = MuiControls.dp(context, 8);
        params.gravity = Gravity.CENTER_VERTICAL;
        bar.addView(button, params);
    }

    private void refreshActions() {
        boolean dirty = this.draft.dirty();

        if (this.applyButton != null) this.applyButton.setEnabled(!this.applying && dirty);
        if (this.revertButton != null) this.revertButton.setEnabled(!this.applying && dirty);

        if (this.doneButton != null) {
            this.doneButton.setEnabled(!this.applying);
            this.doneButton.setText(I18n.get(dirty ? "narutoloading.action.doneApply" : "narutoloading.action.done"));
        }

        if (this.statusView != null) {
            this.statusView.setText(this.applying ? I18n.get("narutoloading.status.applying") : dirty ? I18n.get("narutoloading.status.pending") : "");
        }
    }

    private void finish() {
        if (this.applying) return;

        if (!this.draft.dirty()) {
            this.close();
            return;
        }

        this.apply(true);
    }

    private void apply(boolean closeAfter) {
        if (this.applying) return;

        if (!this.draft.dirty()) {
            if (closeAfter) this.close();
            return;
        }

        boolean playback = this.draft.playbackChanged();

        this.applying = true;
        this.refreshActions();

        Minecraft minecraft = Minecraft.getInstance();
        Runnable task = () -> this.applyOnRenderThread(playback, closeAfter);

        if (minecraft.isSameThread()) {
            task.run();
        } else {
            minecraft.execute(task);
        }
    }

    private void applyOnRenderThread(boolean playback, boolean closeAfter) {
        Throwable failure = null;
        double seekTo = playback ? ClientUiControl.positionSec() : 0D;

        try {
            this.draft.write();
            this.applyVolume();
            ClientUiControl.applyMode();

            if (playback) {
                this.context.media().refresh();
                ClientUiControl.restart(seekTo);
            }
        } catch (Throwable throwable) {
            failure = throwable;
            System.err.println("[NarutoLoading] Exception applying the config: " + throwable.getMessage());
        }

        Throwable error = failure;

        MuiModApi.postToUiThread(() -> {
            this.applying = false;
            this.refreshActions();

            MuiScreenHost current = this.host;

            if (error != null) {
                if (current != null) MuiDialog.notice(current, I18n.get("narutoloading.apply.failed"), failureMessage(error), I18n.get("narutoloading.ok"), () -> {});
                return;
            }

            if (closeAfter) {
                this.close();
                return;
            }

            if (current != null) current.reloadAllPages();

            if (this.statusView != null) {
                this.statusView.setText(I18n.get("narutoloading.status.applied"));
                this.statusView.postDelayed(this::refreshActions, 1600L);
            }
        });
    }

    private void revert() {
        if (this.applying) return;

        this.draft = NarutoDraft.load();
        this.applyVolume();
        MuiShimmer.refresh();

        MuiScreenHost current = this.host;
        if (current != null) current.reloadAllPages();

        this.refreshActions();
    }

    private void close() {
        MuiScreenHost current = this.host;
        if (current != null) current.close();
    }

    private boolean onCloseRequested() {
        if (this.applying) return false;
        if (!this.draft.dirty()) return true;

        MuiScreenHost current = this.host;
        if (current == null) return true;

        if (Core.isOnUiThread()) {
            this.showCloseDialog(current);
        } else {
            MuiModApi.postToUiThread(() -> this.showCloseDialog(current));
        }

        return false;
    }

    private void showCloseDialog(@NotNull MuiScreenHost host) {
        if (this.closeDialogOpen || this.host != host) return;

        this.closeDialogOpen = true;
        MuiDialog.confirm(host, I18n.get("narutoloading.unsaved.title"), I18n.get("narutoloading.unsaved.message"), I18n.get("narutoloading.unsaved.discard"),
                () -> {
                    this.closeDialogOpen = false;
                    host.close();
                }, I18n.get("narutoloading.unsaved.keep"), () -> this.closeDialogOpen = false);
    }

    private void onDestroyed() {
        this.host = null;
        this.downloads.destroy();
        FadeCenter.clearPreview();

        AudioProducer audio = ClientUiControl.active().audio.get();
        if (audio != null) audio.volume(NarutoConfig.volume());
    }

    private static @NotNull String failureMessage(@NotNull Throwable error) {
        String message = error.getMessage();
        return message == null || message.isBlank() ? I18n.get("narutoloading.apply.failedMessage") : message;
    }

    private final class ScreenShimmerConfig implements MuiShimmerConfig {
        @Override
        public boolean enabled() {
            return NarutoScreen.this.draft.get(NarutoSettings.SHIMMER);
        }

        @Override
        public int fps() {
            return NarutoScreen.this.draft.get(NarutoSettings.SHIMMER_FPS);
        }

        @Override
        public double speed() {
            return NarutoScreen.this.draft.get(NarutoSettings.SHIMMER_SPEED);
        }

        @Override
        public double scale() {
            return NarutoScreen.this.draft.get(NarutoSettings.SHIMMER_SCALE);
        }

        @Override
        public double strength() {
            return NarutoScreen.this.draft.get(NarutoSettings.SHIMMER_STRENGTH);
        }

        @Override
        @Contract(value = " -> new", pure = true)
        public int @NotNull [] ramp() {
            return MuiColorSchemes.NARUTO.shimmerRamp();
        }
    }
}

final class PlaybackPage implements SettingsPageBuilder {
    private final NarutoScreen screen;
    private final NarutoContext context;

    private @Nullable MuiSettingsSection ffmpegSection;

    PlaybackPage(@NotNull NarutoScreen screen, @NotNull NarutoContext context) {
        this.screen = screen;
        this.context = context;
    }

    @Override
    public void build(@NotNull MuiSettingsPage page) {
        MediaRepository media = this.context.media();
        MediaRepository.Media current = media.current();
        List<MediaRepository.Media> all = media.all();

        page.shimmerAll()
                .section("narutoloading.playback.section.current", section -> {
                    section.value("narutoloading.playback.current", current == null ? I18n.get("narutoloading.playback.current.none") : current.name())
                            .value("narutoloading.playback.count", Integer.toString(all.size()))
                            .button("narutoloading.playback.import", true, this::importMedia);

                    if (!all.isEmpty()) {
                        List<MuiOption<String>> options = new ArrayList<>(all.size());
                        for (MediaRepository.Media entry : all) options.add(MuiOption.of(entry.name(), entry.name()));

                        section.dropdown("narutoloading.playback.pick", "narutoloading.playback.pick.desc", options, current == null ? null : current.name(), this::selectMedia);
                    }

                    section.button("narutoloading.playback.next", false, this::nextMedia)
                            .button("narutoloading.playback.restart", true, this::restartMedia)
                            .note("narutoloading.playback.note", "narutoloading.playback.note.desc");
                })

                .section("narutoloading.playback.section.audio", section -> section
                        .slider("narutoloading.playback.volume", "narutoloading.playback.volume.desc", 0, 100, () -> Math.round(this.screen.draft().get(NarutoSettings.AUDIO_VOLUME) * 100.0F), value -> this.screen.edit(() -> {
                            this.screen.draft().set(NarutoSettings.AUDIO_VOLUME, value / 100.0F);
                            this.screen.applyVolume();
                        }), value -> value + "%")
                        .toggle("narutoloading.playback.mute", "narutoloading.playback.mute.desc", () -> this.screen.draft().get(NarutoSettings.MUTE_MUSIC), value -> this.screen.edit(() -> this.screen.draft().set(NarutoSettings.MUTE_MUSIC, value))))
                .section("narutoloading.playback.section.ffmpeg", section -> {
                    this.ffmpegSection = section;
                    this.fillFfmpeg(section);
                });
    }

    private void fillFfmpeg(@NotNull MuiSettingsSection section) {
        String ffmpeg = this.context.tools().ffmpeg.ffmpeg.get();
        String ffprobe = this.context.tools().ffmpeg.ffprobe.get();

        section.value("narutoloading.playback.ffmpeg", ffmpeg == null ? I18n.get("narutoloading.playback.ffmpeg.none") : ffmpeg)
                .value("narutoloading.playback.ffprobe", ffprobe == null ? I18n.get("narutoloading.playback.ffmpeg.none") : ffprobe)
                .customLayout(row -> {
                    Context context = row.getContext();
                    MuiTheme theme = row.theme();

                    Button choose = MuiControls.button(context, theme, "narutoloading.playback.ffmpeg.choose", true, view -> this.chooseFfmpegFolder());
                    Button redetect = MuiControls.button(context, theme, "narutoloading.playback.ffmpeg.redetect", true, view -> this.redetectFfmpeg());

                    row.add(MuiLayout.row(context, theme).padding(0).spacing(8).addWeighted(choose, 1.0F).addWeighted(redetect, 1.0F), MuiControls.fill());
                })
                .note("narutoloading.playback.ffmpeg.note", "narutoloading.playback.ffmpeg.note.desc");
    }

    private void chooseFfmpegFolder() {
        MuiFilePicker.openFolder(I18n.get("narutoloading.playback.ffmpeg.choose"), path -> {
            this.context.useFfmpegDirectory(path);
            this.refreshFfmpeg();
        });
    }

    private void redetectFfmpeg() {
        this.context.reloadFFmpeg();
        this.refreshFfmpeg();
    }

    private void refreshFfmpeg() {
        MuiSettingsSection section = this.ffmpegSection;
        if (section != null) section.rebuild(this::fillFfmpeg);
    }

    private void selectMedia(@NotNull String name) {
        this.switchPlayback(() -> this.context.media().select(name));
    }

    private void nextMedia() {
        this.switchPlayback(() -> this.context.media().advance());
    }

    private void restartMedia() {
        this.switchPlayback(() -> {});
    }

    private void switchPlayback(@NotNull Runnable selection) {
        Minecraft minecraft = Minecraft.getInstance();
        minecraft.execute(() -> {
            selection.run();
            ClientUiControl.restart(0D);
            MuiModApi.postToUiThread(this.screen::reloadPage);
        });
    }

    private void importMedia() {
        MuiFilePicker.openFile(I18n.get("narutoloading.playback.import"), I18n.get("narutoloading.playback.import.filter"), this::installMedia, MediaNaming.fileFilters());
    }

    private void installMedia(@NotNull Path source) {
        this.context.media().importMedia(source).whenComplete((target, throwable) -> {
            if (throwable != null) System.err.println("[NarutoLoading] Exception importing " + source + ": " + throwable.getMessage());

            if (target == null) {
                this.screen.notifyImport(I18n.get("narutoloading.playback.import.failed.desc"));
                return;
            }

            String name = MediaNaming.mediaName(target.getFileName().toString());
            this.context.media().refresh(() -> this.selectImported(name));
        });
    }

    private void selectImported(@NotNull String name) {
        Minecraft.getInstance().execute(() -> {
            boolean selected = this.context.media().select(name);
            if (selected) ClientUiControl.restart(0D);

            MuiModApi.postToUiThread(() -> {
                if (!selected) {
                    this.screen.notifyImport(I18n.get("narutoloading.playback.import.unplayable", name));
                    return;
                }

                this.screen.reloadPage();
            });
        });
    }
}

                .value("narutoloading.ytdlp.status.path", available ? this.executable : I18n.get("narutoloading.ytdlp.status.path.none"));

        if (!available) section.note("narutoloading.ytdlp.status.hint", "narutoloading.ytdlp.status.hint.desc");

        section.customLayout(row -> {
            Context context = row.getContext();
            MuiTheme theme = row.theme();

            Button choose = MuiControls.button(context, theme, "narutoloading.ytdlp.action.choose", true, view -> this.chooseExecutable());
            Button redetect = MuiControls.button(context, theme, "narutoloading.ytdlp.action.redetect", true, view -> this.redetect());

            row.add(MuiLayout.row(context, theme).padding(0).spacing(8).addWeighted(choose, 1.0F).addWeighted(redetect, 1.0F), MuiControls.fill());
        });
    }

    private void chooseExecutable() {
        MuiFilePicker.openFile(I18n.get("narutoloading.ytdlp.action.choose"), "yt-dlp", this::installExecutable, ToolRegistry.YTDLP_FILE_FILTER);
    }

    private void installExecutable(@NotNull Path source) {
        if (!Files.isRegularFile(source)) {
            this.publishStatus(I18n.get("narutoloading.ytdlp.status.install.invalid", source.getFileName().toString()), true);
            return;
        }

        String installed = this.tools.installYtDlp(source);

        if (installed == null) {
            this.executable = null;
            this.refreshStatus();
            this.publishStatus(I18n.get("narutoloading.ytdlp.status.install.failed"), true);
            return;
        }

        Path parent = source.getParent();
        if (parent != null) {
            NarutoConfig.set(NarutoConfig.YTDLP_DIR, parent.toString());
            NarutoConfig.save();
            installed = this.tools.reloadYtDlp();
        }

        this.executable = installed;
        this.refreshStatus();

        this.publishStatus(I18n.get("narutoloading.ytdlp.status.install.done", source.getFileName().toString()), false);
    }

    private void redetect() {
        this.executable = this.tools.reloadYtDlp();
        this.refreshStatus();

        this.publishStatus(I18n.get(this.executable != null ? "narutoloading.ytdlp.status.redetect.found" : "narutoloading.ytdlp.status.redetect.missing"), this.executable == null);
    }

    private void refreshStatus() {
        MuiSettingsSection section = this.statusSection;
        if (section != null) section.rebuild(this::fillStatus);
    }

    private void start(boolean video) {
        if (this.executable == null) return;

        String url = this.currentLink();
        String name = this.currentName();
        if (url.isEmpty() || !isValidName(name)) return;

        DownloadState state = this.downloads.state();
        if (!state.begin()) return;

        state.status(I18n.get(video ? "narutoloading.ytdlp.status.start.video" : "narutoloading.ytdlp.status.start.audio"), false);

        Path outputDirectory = this.outputDirectory();
        CompletableFuture<@Nullable String> future = video ? this.downloads.downloadVideo(url, outputDirectory, name, this::onOutputLine) : this.downloads.downloadAudio(url, outputDirectory, name, this::onOutputLine);

        future.whenComplete((result, throwable) -> {
            state.end();

            if (throwable != null) {
                System.err.println("[NarutoLoading] Exception downloading " + url + ": " + throwable.getMessage());
                state.status(I18n.get("narutoloading.ytdlp.status.failed", rootMessage(throwable)), true);
                return;
            }

            if (result == null) {
                state.status(I18n.get("narutoloading.ytdlp.status.failed", I18n.get("narutoloading.ytdlp.status.failed.detail")), true);
                return;
            }

            state.done(I18n.get("narutoloading.ytdlp.status.done", Path.of(result).getFileName().toString()));

            String mediaName = MediaNaming.mediaName(result);
            this.media.refresh(() -> this.play(mediaName));
            this.postNameWarning();
        });
    }

    private void play(@NotNull String name) {
        Minecraft.getInstance().execute(() -> {
            if (this.media.select(name)) {
                ClientUiControl.restart(0D);
                return;
            }

            this.publishStatus(I18n.get("narutoloading.ytdlp.status.unplayable", name), true);
        });
    }

    private void onOutputLine(@NotNull String line) {
        String trimmed = line.trim();
        if (trimmed.isEmpty()) return;

        this.downloads.state().line(trimmed);
    }

    private void onStateChanged() {
        if (!this.renderPending.compareAndSet(false, true)) return;

        TextView view = this.statusView != null ? this.statusView : this.logView;

        if (view == null) {
            this.renderPending.set(false);
            return;
        }

        view.post(() -> {
            this.renderPending.set(false);
            this.renderState();
        });
    }

    private void renderState() {
        TextView statusView = this.statusView;

        if (statusView != null) {
            MuiTheme theme = this.theme;
            statusView.setText(this.currentStatus());
            if (theme != null) statusView.setTextColor(this.currentStatusColor(theme));
        }

        TextView logView = this.logView;
        if (logView != null) logView.setText(this.currentLog());
    }

    private void publishStatus(@NotNull String text, boolean failed) {
        this.downloads.state().status(text, failed);
    }

    private @NotNull String currentStatus() {
        DownloadState state = this.downloads.state();
        if (!state.status().isEmpty()) return state.status();
        return I18n.get(state.downloading() ? "narutoloading.ytdlp.status.working" : "narutoloading.ytdlp.status.idle");
    }

    private int currentStatusColor(@NotNull MuiTheme theme) {
        return this.downloads.state().statusFailed() ? theme.error() : theme.onSurface();
    }

    private @NotNull String currentLog() {
        String log = this.downloads.state().logText();
        return log.isEmpty() ? I18n.get("narutoloading.ytdlp.progress.log.empty") : log;
    }

    private boolean canDownload() {
        return this.executable != null && !this.downloads.state().downloading() && !this.currentLink().isEmpty() && isValidName(this.currentName());
    }

    private @NotNull String currentLink() {
        MuiTextField field = this.linkField;
        String text = field == null ? this.link : field.text();
        return text == null ? "" : text.trim();
    }

    private @NotNull String currentName() {
        MuiTextField field = this.nameField;
        String text = field == null ? this.fileName : field.text();
        return text == null ? "" : text.trim();
    }

    private void refreshNameWarning() {
        TextView view = this.nameWarning;
        if (view == null) return;

        String name = this.currentName();
        String summary = isValidName(name) ? this.collisionSummary(name) : null;

        view.setText(summary == null ? "" : summary);
        view.setVisibility(summary == null ? View.GONE : View.VISIBLE);
    }

    private void postNameWarning() {
        TextView view = this.nameWarning;
        if (view != null) view.post(this::refreshNameWarning);
    }

    private @Nullable String collisionSummary(@NotNull String name) {
        Path outputDirectory = this.outputDirectory();
        if (!Files.isDirectory(outputDirectory)) return null;

        List<String> matches = new ArrayList<>();
        try (Stream<Path> stream = Files.list(outputDirectory)) {
            stream.filter(Files::isRegularFile)
                    .map(path -> path.getFileName().toString())
                    .filter(MediaNaming::isMediaFile)
                    .filter(fileName -> MediaNaming.wouldOverwrite(name, fileName))
                    .sorted().forEach(matches::add);
        } catch (IOException | UncheckedIOException exception) {
            return null;
        }

        return matches.isEmpty() ? null : I18n.get("narutoloading.ytdlp.name.conflict", String.join(", ", matches));
    }

    private void openFolder() {
        Path outputDirectory = this.outputDirectory();
        try {
            Files.createDirectories(outputDirectory);
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception creating " + outputDirectory + ": " + exception.getMessage());
        }

        try {
            Util.getPlatform().openFile(outputDirectory.toFile());
        } catch (Throwable throwable) {
            System.err.println("[NarutoLoading] Exception opening " + outputDirectory + ": " + throwable.getMessage());
        }
    }

    private @NotNull Path outputDirectory() {
        return GamePaths.resolve(NarutoConfig.sourcesDir());
    }

    private static boolean looksLikeUrl(@NotNull String value) {
        String trimmed = value.trim();
        return trimmed.isEmpty() || trimmed.startsWith("http://") || trimmed.startsWith("https://");
    }

    private static boolean isValidName(@NotNull String value) {
        String trimmed = value.trim();
        if (trimmed.isEmpty() || trimmed.equals(".") || trimmed.equals("..")) return false;

        for (int i = 0; i < trimmed.length(); i++) {
            char c = trimmed.charAt(i);
            if (c < ' ' || "/\\:*?\"<>|".indexOf(c) >= 0) return false;
        }

        return true;
    }

    private static @NotNull String rootMessage(@NotNull Throwable throwable) {
        Throwable cause = throwable.getCause();
        String message = (cause == null ? throwable : cause).getMessage();
        return message == null || message.isBlank() ? throwable.getClass().getSimpleName() : message;
    }
}

package me.kall.narutoloading.client.util;

import me.kall.narutoloading.Integration;
import me.kall.narutoloading.client.ui.NarutoScreen;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.components.EditBox;
import net.minecraft.client.gui.screens.Screen;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import static org.lwjgl.glfw.GLFW.*;

public final class PasteListener {
    private static final String MODERN_UI_SCREEN_PACKAGE = "icyllis.modernui.";

    private static int cooldown = 20;

    private PasteListener() {}

    public static void tick() {
        cooldown--;
        if (cooldown > 0) return;

        Minecraft minecraft = Minecraft.getInstance();
        long window = minecraft.getWindow().getWindow();

        boolean pressed = (glfwGetKey(window, GLFW_KEY_LEFT_CONTROL) == GLFW_PRESS || glfwGetKey(window, GLFW_KEY_RIGHT_CONTROL) == GLFW_PRESS) && glfwGetKey(window, GLFW_KEY_V) == GLFW_PRESS;

        if (!pressed) return;

        if (!Integration.NARUTO_UI) return;

        Screen current = minecraft.screen;
        if (hasTextInput(current)) return;

        String url = clipboardUrl(minecraft);
        if (url == null) return;

        cooldown = 20;

        minecraft.setScreen(NarutoScreen.create(current, url));
    }

    private static boolean hasTextInput(@Nullable Screen screen) {
        if (screen == null) return false;
        if (screen.getFocused() instanceof EditBox) return true;
        return screen.getClass().getName().startsWith(MODERN_UI_SCREEN_PACKAGE);
    }

    private static @Nullable String clipboardUrl(@NotNull Minecraft minecraft) {
        String trimmed = minecraft.keyboardHandler.getClipboard().trim();
        return looksLikeUrl(trimmed) ? trimmed : null;
    }

    private static boolean looksLikeUrl(@NotNull String text) {
        if (!text.startsWith("http://") && !text.startsWith("https://")) return false;

        for (int i = 0; i < text.length(); i++) {
            if (Character.isWhitespace(text.charAt(i))) return false;
        }

        return true;
    }
}

package me.kall.narutoloading.config.api;

import org.jetbrains.annotations.Contract;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;
import org.jetbrains.annotations.UnmodifiableView;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.Reader;
import java.io.StringReader;
import java.io.StringWriter;
import java.io.UncheckedIOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.AtomicMoveNotSupportedException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.NoSuchElementException;
import java.util.Properties;
import java.util.Set;

/**
 * 零依赖的 .properties 配置库，支持链式调用，并与 System 属性双向同步。
 *
 * <p>只使用 JDK 8 的基础 API（无 lambda / stream / 日志框架），
 * 因此可以在 Java Agent 的 premain 阶段安全使用。</p>
 *
 * <pre>{@code
 * Config cfg = Config.builder()
 *         .file("config/app.properties")
 *         .prefix("app.")
 *         .defaultValue("port", 8080)
 *         .defaultValue("debug", false)
 *         .autoCreate(true)
 *         .build();
 *
 * int port = cfg.getInt("port");
 * cfg.set("debug", true).set("name", "demo").save();
 * }</pre>
 *
 * <h3>读取优先级</h3>
 * 开启同步时：System 属性（如 -Dapp.port=1） &gt; 配置文件 &gt; 默认值。
 * 调用 {@link Builder#fileOverridesSystem(boolean)} 可让文件值在加载时覆盖已有的 System 属性。
 *
 * <h3>同步规则</h3>
 * <ul>
 *   <li>加载时：文件/默认值中的每个键会写入 System 属性（键为 prefix + key）。</li>
 *   <li>读取时：优先读 System 属性，所以其他代码调用 System.setProperty 也能立即生效。</li>
 *   <li>set / remove 时：同时更新内部存储与 System 属性。</li>
 *   <li>save 只保存内部存储，不会把 -D 临时覆盖值写入文件。</li>
 * </ul>
 */
public final class Config {
    private final Path file;
    private final String prefix;
    private final boolean syncSystem;
    private final boolean fileOverridesSystem;
    private final boolean autoCreate;
    private final String header;
    private final Map<String, String> defaults;
    private final Map<String, String> values = new LinkedHashMap<>();

    private Config(@NotNull Builder builder) {
        this.file = builder.file;
        this.prefix = builder.prefix;
        this.syncSystem = builder.syncSystem;
        this.fileOverridesSystem = builder.fileOverridesSystem;
        this.autoCreate = builder.autoCreate;
        this.header = builder.header;
        this.defaults = new LinkedHashMap<>(builder.defaults);
    }

    /** 创建一个新的配置构建器。 */
    @Contract(" -> new")
    public static @NotNull Builder builder() {
        return new Builder();
    }

    /**
     * 从配置文件加载值，补入缺失的默认值，并按配置同步 System 属性。
     *
     * @return 当前配置实例
     * @throws UncheckedIOException 配置文件读取失败时抛出
     */
    public synchronized Config load() {
        boolean fileExisted = this.file != null && Files.isRegularFile(this.file);
        this.values.clear();

        if (fileExisted) {
            Properties p = new Properties();
            try (Reader r = Files.newBufferedReader(this.file, StandardCharsets.UTF_8)) {
                p.load(r);
            } catch (IOException e) {
                throw new UncheckedIOException("Failed to read config file: " + this.file, e);
            }
            List<String> keys = new ArrayList<>(p.stringPropertyNames());
            Collections.sort(keys);
            for (String k : keys) {
                this.values.put(k, p.getProperty(k));
            }
        }

        for (Map.Entry<String, String> e : this.defaults.entrySet()) {
            if (!this.values.containsKey(e.getKey())) {
                this.values.put(e.getKey(), e.getValue());
            }
        }

        if (this.syncSystem) {
            for (Map.Entry<String, String> e : this.values.entrySet()) {
                if (this.fileOverridesSystem || this.readSys(e.getKey()) == null) {
                    this.writeSys(e.getKey(), e.getValue());
                }
            }
        }

        return !fileExisted && this.autoCreate && this.file != null ? this.save() : this;
    }

    /**
     * 将内部存储的值保存到配置文件。
     *
     * @return 当前配置实例
     * @throws IllegalStateException 未配置文件路径时抛出
     * @throws UncheckedIOException 写入配置文件失败时抛出
     */
    public synchronized Config save() {
        if (this.file == null) {
            throw new IllegalStateException("No file configured; use Builder.file(...)");
        }
        StringBuilder out = new StringBuilder();
        if (this.header != null && !this.header.isEmpty()) {
            for (String line : this.header.split("\\r?\\n", -1)) {
                out.append("# ").append(line).append('\n');
            }
        }
        for (Map.Entry<String, String> e : this.values.entrySet()) {
            out.append(encodeEntry(e.getKey(), e.getValue()));
        }

        Path abs = this.file.toAbsolutePath();
        Path tmp = null;
        try {
            Path parent = abs.getParent();
            Files.createDirectories(parent);
            tmp = Files.createTempFile(parent, ".config", ".tmp");
            Files.writeString(tmp, out.toString());
            try {
                Files.move(tmp, abs, StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
            } catch (AtomicMoveNotSupportedException ex) {
                Files.move(tmp, abs, StandardCopyOption.REPLACE_EXISTING);
            }
        } catch (IOException e) {
            throw new UncheckedIOException("Failed to write config file: " + this.file, e);
        } finally {
            if (tmp != null) {
                try {
                    Files.deleteIfExists(tmp);
                } catch (IOException ignored) {
                    // 临时文件清理失败不影响结果
                }
            }
        }
        return this;
    }

    /** 借助 Properties.store 做转义，保证与标准格式完全兼容；去掉它自带的时间戳注释行。 */
    private static @NotNull String encodeEntry(String key, String value) {
        Properties tmp = new Properties();
        tmp.setProperty(key, value);
        StringWriter sw = new StringWriter();
        try {
            tmp.store(sw, null);
            BufferedReader br = new BufferedReader(new StringReader(sw.toString()));
            br.readLine(); // 跳过 "#<date>"
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) {
                if (!line.isEmpty()) {
                    sb.append(line).append('\n');
                }
            }
            return sb.toString();
        } catch (IOException e) {
            throw new UncheckedIOException(e); // StringWriter 不会真的抛出
        }
    }

    // ------------------------------------------------------------------ 写入（链式）

    /**
     * 设置配置值；对象会通过 {@link String#valueOf(Object)} 转为字符串，{@code null} 等同于移除键。
     *
     * @param key 配置键
     * @param value 配置值
     * @return 当前配置实例
     * @throws IllegalArgumentException 键为 {@code null} 或空字符串时抛出
     */
    public synchronized Config set(String key, Object value) {
        checkKey(key);
        if (value == null) {
            return this.remove(key);
        }
        String s = String.valueOf(value);
        this.values.put(key, s);
        if (this.syncSystem) {
            this.writeSys(key, s);
        }
        return this;
    }

    /**
     * 仅当键不存在时设置配置值。
     *
     * @param key 配置键
     * @param value 配置值
     * @return 当前配置实例
     */
    public synchronized Config setIfAbsent(String key, Object value) {
        return this.has(key) ? this : this.set(key, value);
    }

    /**
     * 移除配置键及其对应的 System 属性。
     *
     * @param key 配置键
     * @return 当前配置实例
     * @throws IllegalArgumentException 键为 {@code null} 或空字符串时抛出
     */
    public synchronized Config remove(String key) {
        checkKey(key);
        this.values.remove(key);
        if (this.syncSystem) {
            try {
                System.clearProperty(this.prefix + key);
            } catch (SecurityException ignored) {
                // 无权限则忽略
            }
        }
        return this;
    }

    // ------------------------------------------------------------------ 读取

    /**
     * 检查配置中是否存在指定键。
     *
     * @param key 配置键
     * @return 键存在时为 {@code true}
     */
    public synchronized boolean has(String key) {
        return this.getString(key) != null;
    }

    /**
     * 获取字符串值；若开启 System 属性同步，则优先返回对应的 System 属性。
     *
     * @param key 配置键
     * @return 配置值，不存在时返回 {@code null}
     */
    public synchronized String getString(String key) {
        checkKey(key);
        if (this.syncSystem) {
            String s = this.readSys(key);
            if (s != null) {
                return s;
            }
        }
        return this.values.get(key);
    }

    /**
     * 获取字符串值；值不存在时返回指定默认值。
     *
     * @param key 配置键
     * @param def 值不存在时返回的默认值
     * @return 配置值或默认值
     */
    public String getString(String key, String def) {
        String v = this.getString(key);
        return v != null ? v : def;
    }

    /**
     * 获取整数值；缺失或格式错误时返回指定默认值。
     *
     * @param key 配置键
     * @param def 缺失或格式错误时返回的默认值
     * @return 解析出的整数或默认值
     */
    public int getInt(String key, int def) {
        String v = this.trimmed(key);
        if (v == null) {
            return def;
        }
        try {
            return Integer.parseInt(v);
        } catch (NumberFormatException e) {
            return def;
        }
    }

    /**
     * 获取长整数值；缺失或格式错误时返回指定默认值。
     *
     * @param key 配置键
     * @param def 缺失或格式错误时返回的默认值
     * @return 解析出的长整数或默认值
     */
    public long getLong(String key, long def) {
        String v = this.trimmed(key);
        if (v == null) {
            return def;
        }
        try {
            return Long.parseLong(v);
        } catch (NumberFormatException e) {
            return def;
        }
    }

    /**
     * 获取双精度浮点值；缺失或格式错误时返回指定默认值。
     *
     * @param key 配置键
     * @param def 缺失或格式错误时返回的默认值
     * @return 解析出的浮点值或默认值
     */
    public double getDouble(String key, double def) {
        String v = this.trimmed(key);
        if (v == null) {
            return def;
        }
        try {
            return Double.parseDouble(v);
        } catch (NumberFormatException e) {
            return def;
        }
    }

    /**
     * 获取布尔值，接受 true/false、yes/no、on/off、1/0（不区分大小写）。
     *
     * @param key 配置键
     * @param def 缺失或格式错误时返回的默认值
     * @return 解析出的布尔值或默认值
     */
    public boolean getBoolean(String key, boolean def) {
        Boolean b = parseBoolean(this.trimmed(key));
        return b != null ? b : def;
    }

    /**
     * 获取必需的字符串值。
     *
     * @param key 配置键
     * @return 配置值
     * @throws NoSuchElementException 键不存在时抛出
     */
    public @NotNull String requireString(String key) {
        String v = this.getString(key);

package me.kall.narutoloading.core;

import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.List;
import java.util.function.Consumer;

public final class Executable {
    private Executable() {}

    public static @Nullable String runCommand(@Nullable Consumer<String> outputLineConsumer, @Nullable List<String> command) {
        if (command == null || command.isEmpty()) return null;

        Process process;

        try {
            process = new ProcessBuilder(command).redirectErrorStream(true).start();
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception starting command " + Arrays.toString(command.toArray()) + " : " + exception.getMessage());
            exception.printStackTrace(System.err);
            return null;
        }

        StringBuilder output = new StringBuilder();
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()))){
            String line;
            while ((line = reader.readLine()) != null) {
                if (outputLineConsumer != null) outputLineConsumer.accept(line);
                output.append(line.trim()).append("\n");
            }
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception consuming input stream of command " + Arrays.toString(command.toArray()) + " : " + exception.getMessage());
            exception.printStackTrace(System.err);
            return null;
        }

        try {
            int exitCode = process.waitFor();
            if (exitCode != 0) {
                System.err.println("[NarutoLoading] Command " + Arrays.toString(command.toArray()) + " exited with code " + exitCode + ". Read log for details.");
                return null;
            }
        } catch (InterruptedException exception) {
            System.err.println("[NarutoLoading] The execution thread of command " + Arrays.toString(command.toArray()) + " is interrupted: " + exception.getMessage());
            exception.printStackTrace(System.err);
            Thread.currentThread().interrupt();
            return null;
        }

        return output.toString().trim();
    }

    public static @Nullable String runCommand(String @NotNull ... command) {
        return runCommand(null, command);
    }

    public static @Nullable String runCommand(@Nullable Consumer<String> outputLineConsumer, String @NotNull ... command) {
        if (command.length == 0) return null;
        return runCommand(outputLineConsumer, List.of(command));
    }
}

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.stream.Stream;

public final class Locator {
    static final boolean WINDOWS = File.separatorChar == '\\';
    private static final int SEARCH_DEPTH = 4;

    private Locator() {}

    static final class Index {
        private final Map<Path, Map<String, String>> directories = new HashMap<>();

        private @Nullable String find(@NotNull Path directory, @NotNull String target) {
            Map<String, String> entries = this.directories.get(directory);
            if (entries == null) {
                entries = walk(directory);
                this.directories.put(directory, entries);
            }

            for (String name : fileNames(target)) {
                String located = entries.get(key(name));
                if (located != null) return located;
            }

            return null;
        }

        private static @NotNull Map<String, String> walk(@NotNull Path directory) {
            Map<String, String> entries = new HashMap<>();
            if (!Files.isDirectory(directory)) return entries;

            try (Stream<Path> stream = Files.walk(directory, SEARCH_DEPTH)) {
                stream.filter(Files::isRegularFile)
                        .forEach(path -> entries.putIfAbsent(key(path.getFileName().toString()), path.toAbsolutePath().toString()));
            } catch (IOException | UncheckedIOException exception) {
                System.err.println("[NarutoLoading] Exception searching " + directory + ": " + exception.getMessage());
            }

            return entries;
        }

        private static @NotNull String key(@NotNull String fileName) {
            return WINDOWS ? fileName.toLowerCase(Locale.ROOT) : fileName;
        }
    }

    public static @Nullable String locate(@NotNull String target, @NotNull Path gameDir, @Nullable String configuredDir) {
        return locate(target, gameDir, null, configuredDir == null || configuredDir.isBlank() ? null : Path.of(configuredDir), new Index());
    }

    static @Nullable String locate(@NotNull String target, @NotNull Path gameDir, @Nullable String neighbor, @Nullable Path configuredDir, @NotNull Index index) {
        String located = null;

        if (configuredDir != null && !configuredDir.toString().isEmpty()) {
            if (configuredDir.isAbsolute()) {
                located = index.find(configuredDir, target);
            } else {
                System.err.println("[NarutoLoading] Configured directory for " + target + " must be absolute: " + configuredDir);
            }
        }

        if (located == null) located = findInPath(target);

        if (located == null && neighbor != null) {
            File parent = new File(neighbor).getParentFile();
            located = parent == null ? null : index.find(parent.toPath(), target);
        }

        if (located == null) located = index.find(gameDir, target);

        if (located == null) {
            System.err.println("[NarutoLoading] Cannot find executable " + target + (configuredDir == null || configuredDir.toString().isEmpty() ? "" : " in " + configuredDir) + ", system PATH, or " + gameDir + " (depth " + SEARCH_DEPTH + ")");
        } else {
            System.out.println("[NarutoLoading] Located executable " + target + " at " + located);
        }

        return located;
    }

    private static @Nullable String findInPath(@NotNull String name) {
        String path = System.getenv("PATH");
        if (path == null) return null;

        for (String entry : path.split(File.pathSeparator)) {
            String dir = entry.trim();
            if (dir.isEmpty()) continue;
            if (dir.length() > 1 && dir.charAt(0) == '"' && dir.charAt(dir.length() - 1) == '"') {
                dir = dir.substring(1, dir.length() - 1);
            }

            String located = findIn(new File(dir), name);
            if (located != null) return located;
        }

        return null;
    }

    private static @Nullable String findIn(@NotNull File directory, @NotNull String name) {
        for (String fileName : fileNames(name)) {
            File candidate = new File(directory, fileName);
            if (candidate.isFile()) return candidate.getAbsolutePath();
        }

        return null;
    }

    private static @NotNull @Unmodifiable List<String> fileNames(@NotNull String name) {
        return WINDOWS ? List.of(name + ".exe", name) : List.of(name);
    }
}

package me.kall.narutoloading.media;

import org.jetbrains.annotations.NotNull;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicBoolean;

public final class DownloadState {
    private static final int LOG_LINES = 5;

    private final AtomicBoolean downloading = new AtomicBoolean();
    private final List<Runnable> listeners = new CopyOnWriteArrayList<>();

    private final Object logLock = new Object();
    private final Deque<String> log = new ArrayDeque<>(LOG_LINES);

    private volatile int percent = -1;
    private volatile String status = "";
    private volatile boolean statusFailed;
    private volatile String logText = "";

    public boolean begin() {
        if (!this.downloading.compareAndSet(false, true)) return false;

        this.percent = -1;
        synchronized (this.logLock) {
            this.log.clear();
            this.logText = "";
        }

        this.notifyListeners();
        return true;
    }

    public void end() {
        this.downloading.set(false);
        this.notifyListeners();
    }

    public void status(@NotNull String text, boolean failed) {
        this.status = text;
        this.statusFailed = failed;
        this.notifyListeners();
    }

    public void done(@NotNull String text) {
        this.percent = 100;
        this.status = text;
        this.statusFailed = false;
        this.notifyListeners();
    }

    public void line(@NotNull String line) {
        int parsed = parsePercent(line);
        if (parsed >= 0) this.percent = parsed;

        synchronized (this.logLock) {
            this.log.addLast(line);
            while (this.log.size() > LOG_LINES) this.log.removeFirst();
            this.logText = String.join("\n", this.log);
        }

        this.notifyListeners();
    }

    public boolean downloading() {
        return this.downloading.get();
    }

    public int percent() {
        return this.percent;
    }

    public @NotNull String status() {
        return this.status;
    }

    public boolean statusFailed() {
        return this.statusFailed;
    }

    public @NotNull String logText() {
        return this.logText;
    }

    public void addListener(@NotNull Runnable listener) {
        this.listeners.add(listener);
    }

    public void removeListener(@NotNull Runnable listener) {
        this.listeners.remove(listener);
    }

    private void notifyListeners() {
        for (Runnable listener : this.listeners) listener.run();
    }

    private static int parsePercent(@NotNull String line) {
        if (!line.startsWith("[download]")) return -1;

        int percentIndex = line.indexOf('%');
        if (percentIndex <= 0) return -1;

        int start = percentIndex - 1;
        while (start >= 0 && (Character.isDigit(line.charAt(start)) || line.charAt(start) == '.')) start--;

        try {
            return (int) Math.round(Double.parseDouble(line.substring(start + 1, percentIndex)));
        } catch (NumberFormatException exception) {
            return -1;
        }
    }
}

package me.kall.narutoloading.media.ffmpeg;

import me.kall.narutoloading.core.Executable;
import me.kall.narutoloading.media.ffmpeg.data.AudioArgs;
import me.kall.narutoloading.media.ffmpeg.data.VideoArgs;
import org.apache.commons.io.FilenameUtils;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.InvalidPathException;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.atomic.AtomicReference;

public final class FFmpeg {
    private static final String VIDEO_ENTRIES = "stream=codec_type,width,height,r_frame_rate,avg_frame_rate,start_time:format=duration";
    private static final String AUDIO_ENTRIES = "stream=codec_type,channels,sample_rate,start_time:format=duration";
    private static final String MONO_ENTRIES = "stream=codec_type,channels";

    private static final String STREAM_CODEC_TYPE = "streams.stream.0.codec_type";
    private static final String STREAM_WIDTH = "streams.stream.0.width";
    private static final String STREAM_HEIGHT = "streams.stream.0.height";
    private static final String STREAM_AVG_FPS = "streams.stream.0.avg_frame_rate";
    private static final String STREAM_R_FPS = "streams.stream.0.r_frame_rate";
    private static final String STREAM_START_TIME = "streams.stream.0.start_time";
    private static final String STREAM_CHANNELS = "streams.stream.0.channels";
    private static final String STREAM_SAMPLE_RATE = "streams.stream.0.sample_rate";
    private static final String FORMAT_DURATION = "format.duration";

    private static final String NOT_AVAILABLE = "N/A";

    public final AtomicReference<@Nullable String> ffmpeg = new AtomicReference<>();
    public final AtomicReference<@Nullable String> ffprobe = new AtomicReference<>();

    public @Nullable VideoArgs deriveVideo(@Nullable String absPathStr) {
        String ffprobe = this.ffprobe.get();
        if (ffprobe == null || absPathStr == null) return null;

        Path absPath = regularFile(absPathStr, "video");
        if (absPath == null) return null;

        Map<String, String> values = this.probe(absPathStr, "v:0", VIDEO_ENTRIES);
        if (values == null) {
            System.err.println("[NarutoLoading] FFprobe " + ffprobe + " returned no data for " + absPath + " or encounters exception during the process.");
            return null;
        }

        if (!"video".equals(values.get(STREAM_CODEC_TYPE))) {
            System.err.println("[NarutoLoading] Video stream is not found in " + absPathStr);
            return null;
        }

        double avg = parseRate(values.get(STREAM_AVG_FPS));
        double fps = avg > 0D && Double.isFinite(avg) ? avg : parseRate(values.get(STREAM_R_FPS));
        if (!(fps > 0D) || !Double.isFinite(fps)) {
            System.err.println("[NarutoLoading] There is no usable frame rate for " + absPathStr);
            return null;
        }

        int width = parseInteger(values, STREAM_WIDTH, -1);
        if (width <= 0) {
            System.err.println("[NarutoLoading] There is no usable width for " + absPathStr);
            return null;
        }

        int height = parseInteger(values, STREAM_HEIGHT, -1);
        if (height <= 0) {
            System.err.println("[NarutoLoading] There is no usable height for " + absPathStr);
            return null;
        }

        double duration = parseDouble(values, FORMAT_DURATION, Double.NaN);
        if (Double.isNaN(duration)) {
            System.err.println("[NarutoLoading] There is no usable duration for " + absPathStr);
            return null;
        }

        double startSec = parseDouble(values, STREAM_START_TIME, 0D);
        return new VideoArgs(absPathStr, fps, width, height, duration, startSec);
    }

    public @Nullable AudioArgs deriveAudio(@Nullable String absPathStr) {
        String ffprobe = this.ffprobe.get();
        if (ffprobe == null || absPathStr == null) return null;

        Path absPath = regularFile(absPathStr, "audio");
        if (absPath == null) return null;

        Map<String, String> values = this.probe(absPathStr, "a:0", AUDIO_ENTRIES);
        if (values == null) {
            System.err.println("[NarutoLoading] FFprobe " + ffprobe + " returned no data for " + absPath + " or encounters exception during the process.");
            return null;
        }

        if (!"audio".equals(values.get(STREAM_CODEC_TYPE))) {
            System.err.println("[NarutoLoading] Audio stream is not found in " + absPathStr);
            return null;
        }

        int channelCount = parseInteger(values, STREAM_CHANNELS, -1);
        if (channelCount <= 0) {
            System.err.println("[NarutoLoading] There is no usable channel count for " + absPathStr);
            return null;
        }

        int sampleRate = parseInteger(values, STREAM_SAMPLE_RATE, -1);
        if (sampleRate <= 0) {
            System.err.println("[NarutoLoading] There is no usable sample rate for " + absPathStr);
            return null;
        }

        if (channelCount > 2) {
            System.err.println("[NarutoLoading] Unsupported OpenAL format for " + channelCount + " channel(s) in " + absPathStr);
            return null;
        }

        double duration = parseDouble(values, FORMAT_DURATION, Double.NaN);
        if (Double.isNaN(duration)) {
            System.err.println("[NarutoLoading] There is no usable duration for " + absPathStr);
            return null;
        }

        double startSec = parseDouble(values, STREAM_START_TIME, 0D);
        return new AudioArgs(absPathStr, channelCount, sampleRate, startSec, duration);
    }

    public @Nullable String monoize(@Nullable String absPathStr) {
        String ffmpeg = this.ffmpeg.get();
        if (absPathStr == null || ffmpeg == null) return null;

        Path absPath = regularFile(absPathStr, "audio");
        if (absPath == null) return null;

        if ("ogg".equalsIgnoreCase(FilenameUtils.getExtension(absPathStr)) && this.monoCheck(absPathStr)) return absPathStr;

        String name = FilenameUtils.getBaseName(absPathStr);

        File absOutput = absPath.getParent().resolve(name + ".ogg").toFile();
        String absOutputStr = absOutput.getAbsolutePath();

        if (absOutput.exists()) {
            if (this.monoCheck(absOutputStr)) return absOutputStr;

            if (!absOutput.delete()) {
                System.err.println("[NarutoLoading] Error deleting stale mono audio output " + absOutputStr);
                return null;
            }
        }

        return Executable.runCommand(ffmpeg, "-i", absPathStr, "-vn", "-acodec", "libvorbis", "-ac", "1", "-q:a", "4", "-y", absOutputStr) != null ? absOutputStr : null;
    }

    private static @Nullable Path regularFile(@NotNull String absPathStr, @NotNull String kind) {
        Path absPath;
        try {
            absPath = Path.of(absPathStr);
        } catch (InvalidPathException exception) {
            System.err.println("[NarutoLoading] " + absPathStr + " path string cannot be converted to a Path object: " + exception.getMessage());
            return null;
        }

        if (!Files.isRegularFile(absPath)) {
            System.err.println("[NarutoLoading] " + absPathStr + " does not exist or is not a valid regular " + kind + " file.");
            return null;
        }

        return absPath;
    }

    private boolean monoCheck(String absPathStr) {
        if (this.ffprobe.get() == null) return false;
        Map<String, String> values = this.probe(absPathStr, "a:0", MONO_ENTRIES);
        return values != null && "audio".equals(values.get(STREAM_CODEC_TYPE)) && parseInteger(values, STREAM_CHANNELS, -1) == 1;
    }

    private @Nullable Map<String, String> probe(@NotNull String absPathStr, @NotNull String selector, @NotNull String entries) {
        String ffprobe = this.ffprobe.get();
        if (ffprobe == null) return null;

        String output = Executable.runCommand(ffprobe, "-v", "quiet", "-select_streams", selector, "-show_entries", entries, "-of", "flat", absPathStr);
        if (output == null || output.isEmpty()) return null;

        Map<String, String> values = new HashMap<>();
        for (String line : output.split("\n")) {
            int separator = line.indexOf('=');
            if (separator <= 0) continue;

            String value = line.substring(separator + 1);
            if (value.length() > 1 && value.charAt(0) == '"' && value.charAt(value.length() - 1) == '"') {
                value = value.substring(1, value.length() - 1);
            }

            values.put(line.substring(0, separator), value);
        }

        return values.isEmpty() ? null : values;
    }

    @SuppressWarnings("SameParameterValue")
    private static int parseInteger(@NotNull Map<String, String> values, @NotNull String key, int fallback) {
        String value = values.get(key);
        if (value == null || value.equals(NOT_AVAILABLE)) return fallback;
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException exception) {
            return fallback;
        }
    }

    private static double parseDouble(@NotNull Map<String, String> values, @NotNull String key, double fallback) {
        String value = values.get(key);
        if (value == null || value.equals(NOT_AVAILABLE)) return fallback;
        try {
            return Double.parseDouble(value);
        } catch (NumberFormatException exception) {
            return fallback;
        }
    }

    private static double parseRate(@Nullable String value) {
        if (value == null || value.equals(NOT_AVAILABLE)) return Double.NaN;

        int slash = value.indexOf('/');
        if (slash <= 0) return Double.NaN;

        try {
            double denominator = Double.parseDouble(value.substring(slash + 1));
            return denominator == 0D ? Double.NaN : Double.parseDouble(value.substring(0, slash)) / denominator;
        } catch (NumberFormatException exception) {
            return Double.NaN;
        }
    }
}

import org.apache.commons.io.FilenameUtils;
import org.jetbrains.annotations.NotNull;

import java.util.ArrayList;

public final class MediaNaming {
    static final List<String> VIDEO_EXTENSIONS = List.of("mp4", "mkv", "webm", "mov", "m4v", "avi", "flv");
    static final List<String> AUDIO_EXTENSIONS = List.of("ogg", "opus", "mp3", "flac", "wav", "m4a", "aac");

    public static final String AUDIO_SUFFIX = ".audio";

    private static final String[] FILE_FILTERS = buildFileFilters();

    private MediaNaming() {}

    public static boolean isMediaFile(@NotNull String fileName) {
        String extension = FilenameUtils.getExtension(fileName).toLowerCase(Locale.ROOT);
        return VIDEO_EXTENSIONS.contains(extension) || AUDIO_EXTENSIONS.contains(extension);
    }

    public static @NotNull String[] fileFilters() {
        return FILE_FILTERS.clone();
    }

    public static @NotNull String mediaName(@NotNull String fileName) {
        String baseName = FilenameUtils.getBaseName(fileName);
        String extension = FilenameUtils.getExtension(fileName).toLowerCase(Locale.ROOT);
        return AUDIO_EXTENSIONS.contains(extension) ? stripAudioSuffix(baseName) : baseName;
    }

    public static boolean wouldOverwrite(@NotNull String name, @NotNull String fileName) {
        String baseName = FilenameUtils.getBaseName(fileName);
        String extension = FilenameUtils.getExtension(fileName).toLowerCase(Locale.ROOT);

        if (VIDEO_EXTENSIONS.contains(extension)) return baseName.equalsIgnoreCase(name.trim());
        if (AUDIO_EXTENSIONS.contains(extension)) return baseName.equalsIgnoreCase(name.trim() + AUDIO_SUFFIX);

        return false;
    }

    private static @NotNull String stripAudioSuffix(@NotNull String baseName) {
        int at = baseName.length() - AUDIO_SUFFIX.length();
        if (at <= 0) return baseName;
        if (!baseName.regionMatches(true, at, AUDIO_SUFFIX, 0, AUDIO_SUFFIX.length())) return baseName;
        return baseName.substring(0, at);
    }

    private static String @NotNull [] buildFileFilters() {
        List<String> extensions = new ArrayList<>(VIDEO_EXTENSIONS.size() + AUDIO_EXTENSIONS.size());
        extensions.addAll(VIDEO_EXTENSIONS);
        extensions.addAll(AUDIO_EXTENSIONS);

        String[] patterns = new String[extensions.size()];
        for (int index = 0; index < patterns.length; index++) patterns[index] = "*." + extensions.get(index);
        return patterns;
    }
}

package me.kall.narutoloading.media;

import me.kall.narutoloading.media.ffmpeg.data.AudioArgs;
import me.kall.narutoloading.media.ffmpeg.data.VideoArgs;
import me.kall.narutoloading.core.GamePaths;
import org.apache.commons.io.FilenameUtils;
import org.jetbrains.annotations.Contract;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.io.IOException;

@SuppressWarnings("UnusedReturnValue")
public final class MediaRepository {
    private static final long IDLE_RESCAN_NANOS = 3_000_000_000L;

    public enum Mode {
        RANDOM, SEQUENTIAL, FIXED;

        public static @NotNull Mode parse(@Nullable String value) {
            if (value == null) return RANDOM;
            try {
                return valueOf(value.trim().toUpperCase(Locale.ROOT));
            } catch (IllegalArgumentException exception) {
                return RANDOM;
            }
        }
    }

    public record Media(@NotNull String name, @Nullable VideoArgs video, @Nullable AudioArgs audio) {}

    private record Probed<T>(long modified, long size, @Nullable T value) {}

    private final Supplier<Path> sourcesDir;
    private final Supplier<Mode> mode;
    private final Supplier<String> fixedName;
    private final Function<String, CompletableFuture<@Nullable VideoArgs>> videoReader;
    private final Function<String, CompletableFuture<@Nullable AudioArgs>> audioReader;
    private final ExecutorService importExecutor;

    private final ExecutorService executor = Executors.newSingleThreadExecutor(task -> {
        Thread thread = new Thread(task, "NarutoMedia");
        thread.setDaemon(true);
        return thread;
    });

    private final AtomicBoolean scanning = new AtomicBoolean();
    private volatile long lastScanStart = 0L;
    private volatile boolean everScanned = false;

    private volatile List<Media> ready = List.of();

    private final Map<Path, Probed<VideoArgs>> videoCache = new ConcurrentHashMap<>();
    private final Map<Path, Probed<AudioArgs>> audioCache = new ConcurrentHashMap<>();

    private final Set<String> warned = ConcurrentHashMap.newKeySet();

    private final Object lock = new Object();

    private @Nullable Media current;
    private @Nullable String lastName;
    private @Nullable String pendingPath;

    public MediaRepository(@NotNull Supplier<Path> sourcesDir, @NotNull Supplier<Mode> mode, @NotNull Supplier<String> fixedName, @NotNull Function<String, CompletableFuture<@Nullable VideoArgs>> videoReader, @NotNull Function<String, CompletableFuture<@Nullable AudioArgs>> audioReader, @NotNull ExecutorService importExecutor) {
        this.sourcesDir = sourcesDir;
        this.mode = mode;
        this.fixedName = fixedName;
        this.videoReader = videoReader;
        this.audioReader = audioReader;
        this.importExecutor = importExecutor;
    }

    public @Nullable Media current() {
        synchronized (this.lock) {
            if (this.current == null) this.select();
            if (this.current == null) this.refresh(false, null);
            return this.current;
        }
    }

    public boolean hasSource() {
        return this.current() != null;
    }

    public @NotNull List<Media> all() {
        return this.ready;
    }

    public boolean select(@NotNull String name) {
        synchronized (this.lock) {
            for (Media media : this.ready) {
                if (media.name().equalsIgnoreCase(name)) {
                    this.current = media;
                    this.lastName = media.name();
                    return true;
                }
            }
        }

        return false;
    }

    public void adopt(@Nullable String absPath) {
        if (absPath == null || absPath.isBlank()) return;

        synchronized (this.lock) {
            for (Media media : this.ready) {
                if (media.video() != null && media.video().absPath().equals(absPath)) {
                    this.current = media;
                    this.lastName = media.name();
                    return;
                }
            }

            this.pendingPath = absPath;
        }
    }

    public @Nullable VideoArgs peekVideo() {
        Media media = this.current();
        return media == null ? null : media.video();
    }

    public @Nullable AudioArgs peekAudio() {
        Media media = this.current();
        return media == null ? null : media.audio();
    }

    public @Nullable Media advance() {
        Media picked;
        synchronized (this.lock) {
            this.select();
            picked = this.current;
        }

        this.refresh();
        return picked;
    }

    @Contract("_ -> new")
    public @NotNull CompletableFuture<@Nullable Path> importMedia(@NotNull Path source) {
        return CompletableFuture.supplyAsync(() -> this.copyMedia(source), this.importExecutor);
    }

    public void refresh() {
        this.refresh(true, null);
    }

    public void refresh(@NotNull Runnable onDone) {
        this.refresh(true, onDone);
    }

    public void invalidate() {
        this.videoCache.clear();
        this.audioCache.clear();

    }

    private void select() {
        List<Media> candidates = new ArrayList<>();
        for (Media media : this.ready) {
            if (exists(media)) candidates.add(media);
        }

        Media picked = MediaSelector.pick(candidates, Media::name, this.mode.get(), this.fixedName.get(), this.lastName, true);

        this.current = picked;
        if (picked != null) this.lastName = picked.name();
    }

    private static boolean exists(@NotNull Media media) {
        VideoArgs video = media.video();
        AudioArgs audio = media.audio();
        if (video != null && !Files.isRegularFile(Path.of(video.absPath()))) return false;
        return audio == null || Files.isRegularFile(Path.of(audio.absPath()));
    }

    private @Nullable Path copyMedia(@NotNull Path source) {
        Path directory = GamePaths.resolve(this.sourcesDir.get());
        Path target = uniqueTarget(directory, source.getFileName().toString());

        try {
            Files.createDirectories(directory);
            if (!source.toAbsolutePath().normalize().equals(target.toAbsolutePath().normalize())) Files.copy(source, target);
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception copying " + source + " to " + target + ": " + exception.getMessage());
            return null;
        }

        return target;
    }

    private static @NotNull Path uniqueTarget(@NotNull Path directory, @NotNull String fileName) {
        Path target = directory.resolve(fileName);
        if (!Files.exists(target)) return target;

        String baseName = FilenameUtils.getBaseName(fileName);
        String extension = FilenameUtils.getExtension(fileName);
        String suffix = extension.isEmpty() ? "" : "." + extension;

        for (int index = 2; ; index++) {
            Path candidate = directory.resolve(baseName + " (" + index + ")" + suffix);
            if (!Files.exists(candidate)) return candidate;
        }
    }

    private void refresh(boolean force, @Nullable Runnable onDone) {
        long now = System.nanoTime();
        if (!force && this.everScanned && now - this.lastScanStart < IDLE_RESCAN_NANOS) return;
        if (!this.scanning.compareAndSet(false, true)) {
            if (onDone != null) this.submit(() -> this.refresh(true, onDone));
            return;
        }

        this.everScanned = true;
        this.lastScanStart = now;

        try {
            this.executor.execute(() -> {
                try {
                    this.rebuild();
                } catch (Throwable throwable) {
                    System.err.println("[NarutoLoading] Exception scanning media sources: " + throwable.getMessage());
                    throwable.printStackTrace(System.err);
                } finally {
                    this.scanning.set(false);
                }

                if (onDone != null) onDone.run();
            });
        } catch (RuntimeException exception) {
            this.scanning.set(false);
        }
    }

    private void submit(@NotNull Runnable task) {
        try {
            this.executor.execute(task);
        } catch (RuntimeException exception) {
            System.err.println("[NarutoLoading] Exception scheduling a media rescan: " + exception.getMessage());
        }
    }

    private void rebuild() {
        List<MediaScanner.Group> groups = MediaScanner.scan(GamePaths.resolve(this.sourcesDir.get()));

        List<Media> result = new ArrayList<>(groups.size());
        Set<Path> seen = new HashSet<>();

        for (MediaScanner.Group group : groups) {
            VideoArgs video = null;
            if (group.video() != null) {
                seen.add(group.video());
                video = probe(this.videoCache, group.video(), this.videoReader);
            }

            Path audioSource = group.audio() != null ? group.audio() : group.video();
            AudioArgs audio = null;
            if (audioSource != null) {
                seen.add(audioSource);
                audio = probe(this.audioCache, audioSource, this.audioReader);
            }

            if (video == null && audio == null) {
                if (this.warned.add("unusable:" + group.name())) System.err.println("[NarutoLoading] Skipping unusable media source '" + group.name() + "'.");
                continue;
            }

            result.add(new Media(group.name(), video, audio));
        }

        this.videoCache.keySet().retainAll(seen);
        this.audioCache.keySet().retainAll(seen);

        this.ready = List.copyOf(result);

        String pending = this.pendingPath;

        if (pending != null) {
            synchronized (this.lock) {
                for (Media media : this.ready) {
                    if (media.video() != null && media.video().absPath().equals(pending)) {
                        this.current = media;
                        this.lastName = media.name();
                        break;
                    }
                }

                this.pendingPath = null;
            }
        }
    }

    public static @Nullable MediaSource pickSource(@NotNull Path directory, @NotNull Mode mode, @Nullable String fixedName) {
        List<MediaScanner.Group> groups = MediaScanner.scan(directory);

        List<MediaScanner.Group> candidates = new ArrayList<>(groups.size());
        for (MediaScanner.Group group : groups) {
            if (group.video() != null) candidates.add(group);
        }

        MediaScanner.Group picked = MediaSelector.pick(candidates, MediaScanner.Group::name, mode, fixedName, null, false);
        if (picked == null) return null;

        Path video = picked.video();
        Path audio = picked.audio();

        if (video == null) return null;

        return new MediaSource(video.toAbsolutePath().toString(), audio == null ? null : audio.toAbsolutePath().toString());
    }

    private static <T> @Nullable T probe(@NotNull Map<Path, Probed<T>> cache, @NotNull Path path, @NotNull Function<String, CompletableFuture<@Nullable T>> reader) {
        long modified;
        long size;
        try {
            modified = Files.getLastModifiedTime(path).toMillis();
            size = Files.size(path);
        } catch (IOException exception) {
            cache.remove(path);
            return null;
        }

        Probed<T> cached = cache.get(path);
        if (cached != null && cached.modified() == modified && cached.size() == size) return cached.value();

        T value;
        try {
            value = reader.apply(path.toAbsolutePath().toString()).join();
        } catch (CompletionException | CancellationException exception) {
            value = null;
        }

        if (value == null) {
            cache.remove(path);
            return null;
        }

        cache.put(path, new Probed<>(modified, size, value));
        return value;
    }
}

package me.kall.narutoloading.media;

import org.apache.commons.io.FilenameUtils;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.io.IOException;
import java.io.UncheckedIOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.TreeMap;
import java.util.stream.Stream;

public final class MediaScanner {
    public record Group(@NotNull String name, @Nullable Path video, @Nullable Path audio) {}

    private MediaScanner() {}

    public static @NotNull List<Group> scan(@NotNull Path dir) {
        if (!Files.isDirectory(dir)) return List.of();

        List<Path> files;
        try (Stream<Path> stream = Files.list(dir)) {
            files = stream.filter(Files::isRegularFile).sorted().toList();
        } catch (IOException | UncheckedIOException exception) {
            System.err.println("[NarutoLoading] Exception listing sources directory " + dir + ": " + exception.getMessage());
            return List.of();
        }

        Map<String, Path[]> slots = new TreeMap<>(String.CASE_INSENSITIVE_ORDER);

        for (Path file : files) {
            String fileName = file.getFileName().toString();
            String extension = FilenameUtils.getExtension(fileName).toLowerCase(Locale.ROOT);

            boolean isVideo = MediaNaming.VIDEO_EXTENSIONS.contains(extension);
            boolean isAudio = MediaNaming.AUDIO_EXTENSIONS.contains(extension);
            if (!isVideo && !isAudio) continue;

            Path[] slot = slots.computeIfAbsent(MediaNaming.mediaName(fileName), key -> new Path[2]);
            if (isVideo) slot[0] = better(slot[0], file, MediaNaming.VIDEO_EXTENSIONS);
            if (isAudio) slot[1] = better(slot[1], file, MediaNaming.AUDIO_EXTENSIONS);
        }

        List<Group> groups = new ArrayList<>(slots.size());
        for (Map.Entry<String, Path[]> entry : slots.entrySet()) {
            groups.add(new Group(entry.getKey(), entry.getValue()[0], entry.getValue()[1]));
        }

        return groups;
    }

    private static @NotNull Path better(@Nullable Path existing, @NotNull Path candidate, @NotNull List<String> priority) {
        if (existing == null) return candidate;
        return rank(candidate, priority) < rank(existing, priority) ? candidate : existing;
    }

    private static int rank(@NotNull Path path, @NotNull List<String> priority) {
        int index = priority.indexOf(FilenameUtils.getExtension(path.getFileName().toString()).toLowerCase(Locale.ROOT));
        return index < 0 ? Integer.MAX_VALUE : index;
    }
}

package me.kall.narutoloading.media;

import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ThreadLocalRandom;
import java.util.function.Function;

public final class MediaSelector {
    private static final Set<String> WARNED = ConcurrentHashMap.newKeySet();

    private MediaSelector() {}

    public static <T> @Nullable T pick(@NotNull List<T> candidates, @NotNull Function<T, String> namer, @NotNull MediaRepository.Mode mode, @Nullable String fixedName, @Nullable String lastName, boolean warnMissingFixed) {
        if (candidates.isEmpty()) return null;
        if (candidates.size() == 1) return candidates.get(0);

        if (mode == MediaRepository.Mode.FIXED) {
            if (fixedName != null && !fixedName.isBlank()) {
                for (T candidate : candidates) {
                    if (namer.apply(candidate).equalsIgnoreCase(fixedName.trim())) return candidate;
                }

                if (warnMissingFixed && WARNED.add("fixed:" + fixedName)) {
                    System.err.println("[NarutoLoading] fixedName '" + fixedName + "' was not found in sources, falling back to random selection.");
                }
            }
            mode = MediaRepository.Mode.RANDOM;
        }

        if (mode == MediaRepository.Mode.SEQUENTIAL) {
            int next = 0;
            for (int i = 0; i < candidates.size(); i++) {
                if (namer.apply(candidates.get(i)).equalsIgnoreCase(lastName)) {
                    next = (i + 1) % candidates.size();
                    break;
                }
            }
            return candidates.get(next);
        }

        List<T> pool = new ArrayList<>(candidates.size());
        for (T candidate : candidates) {
            if (!namer.apply(candidate).equalsIgnoreCase(lastName)) pool.add(candidate);
        }

        if (pool.isEmpty()) pool = candidates;
        return pool.get(ThreadLocalRandom.current().nextInt(pool.size()));
    }
}

    private @Nullable String download(String url, Path outputDirectory, String stem, String format, @Nullable Consumer<String> progress) {
        String ytdlp = this.ytdlp.get();
        if (ytdlp == null) return null;

        Path record;
        try {
            Files.createDirectories(outputDirectory);
            record = Files.createTempFile("narutoloading-ytdlp", ".txt");
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception preparing download. Url: " + url + ". Output Directory: " + outputDirectory + ". Exception: " + exception.getMessage());
            return null;
        }

        try {
            String output = Executable.runCommand(progress, ytdlp, url, "-o", outputDirectory.resolve(template(stem) + ".%(ext)s").toString(), "--format", format, "--no-playlist", "--progress", "--no-simulate", "--force-overwrites", "--print-to-file", "after_move:%(filepath)s", record.toString());

            if (output == null) return null;

            String recorded = lastNonEmptyLine(Files.readString(record));
            if (recorded == null) {
                System.err.println("[NarutoLoading] yt-dlp did not report an output filepath for url: " + url);
                return null;
            }

            return recorded;
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception reading download result for url: " + url + ". Exception: " + exception.getMessage());
            return null;
        } finally {
            deleteStale(record);
        }
    }

    private static @NotNull String template(@NotNull String stem) {
        return stem.replace("%", "%%");
    }

    private static @Nullable String lastNonEmptyLine(@NotNull String text) {
        String last = null;
        for (String line : text.split("\\R")) {
            String trimmed = line.trim();
            if (!trimmed.isEmpty()) last = trimmed;
        }

        return last;
    }

    private static void deleteStale(Path path) {
        try {
            Files.deleteIfExists(path);
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Exception deleting stale " + path + ": " + exception.getMessage());
        }
    }
}

            }

            }
        }

package me.kall.narutoloading.playback;

import me.kall.narutoloading.media.ffmpeg.data.AudioArgs;
import me.kall.narutoloading.media.ffmpeg.data.VideoArgs;
import me.kall.narutoloading.playback.render.FrameRenderer;
import me.kall.narutoloading.config.NarutoConfig;
import me.kall.narutoloading.playback.produce.audio.AudioProducer;
import me.kall.narutoloading.playback.produce.util.TimeLine;
import me.kall.narutoloading.playback.produce.video.AbstractFrameProducer;
import org.jetbrains.annotations.Nullable;

import java.util.concurrent.CompletableFuture;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicReference;

public abstract class AbstractTV<T> {
    private static final long VIDEO_WARM_TIMEOUT_NANOS = 3_000_000_000L;

    public final AtomicReference<@Nullable AbstractFrameProducer<T>> video = new AtomicReference<>();
    public final AtomicReference<@Nullable AudioProducer> audio = new AtomicReference<>();

    public final AtomicReference<@Nullable TimeLine> timeLine = new AtomicReference<>();

    public final AtomicReference<@Nullable CompletableFuture<@Nullable AbstractFrameProducer<T>>> nextVideoTask = new AtomicReference<>();
    public final AtomicReference<@Nullable CompletableFuture<@Nullable AudioProducer>> nextAudioTask = new AtomicReference<>();

    public final FrameRenderer<T> renderer;

    private final AtomicBoolean running = new AtomicBoolean();

    private final AtomicBoolean videoDelayed = new AtomicBoolean();

    private final AtomicLong setupNanos = new AtomicLong();

    public AbstractTV(FrameRenderer<T> renderer) {
        this.renderer = renderer;
    }

    public synchronized boolean isRunning() {
        return this.running.get();
    }

    public abstract boolean prerequisite();

    public abstract @Nullable VideoArgs newVideoArgs();
    public abstract @Nullable AudioArgs newAudioArgs();

    public abstract @Nullable AbstractFrameProducer<T> createVideo(@Nullable VideoArgs videoArgs);
    public abstract @Nullable AudioProducer createAudio(@Nullable AudioArgs audioArgs);

    protected @Nullable TimeLine newTimeLine(@Nullable VideoArgs videoArgs, @Nullable AudioArgs audioArgs, double seekTo) {
        if (videoArgs == null && audioArgs == null) return null;

        TimeLine timeLine = videoArgs != null ? TimeLine.video(videoArgs.duration(), videoArgs.fps()) : TimeLine.audio(audioArgs.duration());

        return timeLine.withPauseListener(this::pauseAudio).withResumeListener(this::resumeAudio).seekTo(seekTo);
    }

    private void pauseAudio() {
        AudioProducer audio = this.audio.get();
        if (audio != null) audio.shutdown();
    }

    private void resumeAudio(double at) {
        AudioProducer audio = this.audio.get();
        if (audio != null) audio.setup(at);
    }

    private @Nullable AbstractFrameProducer<T> buildVideo(@Nullable VideoArgs videoArgs) {
        if (videoArgs == null) return null;
        AbstractFrameProducer<T> video = this.createVideo(videoArgs);
        return video == null ? null : video.withWarmListener(() -> this.startClock(video, null));
    }

    private @Nullable AudioProducer buildAudio(@Nullable AudioArgs audioArgs) {
        if (audioArgs == null) return null;
        AudioProducer audio = this.createAudio(audioArgs);
        return audio == null ? null : audio.withWarmListener(() -> this.startClock(null, audio));
    }

    private void startClock(@Nullable AbstractFrameProducer<T> video, @Nullable AudioProducer audio) {
        if (video != null) {
            if (this.video.get() != video) return;
        } else {
            if (this.video.get() != null && !this.videoDelayed.get()) return;
            if (audio == null || this.audio.get() != audio) return;
        }

        TimeLine timeLine = this.timeLine.get();
        if (timeLine == null) return;

        timeLine.start();

        if (video != null) {
            AudioProducer current = this.audio.get();
            if (current != null) current.eager();
        }
    }

    public synchronized void setup(double seekTo) {
        if (!this.prerequisite()) return;

        this.videoDelayed.set(false);
        this.setupNanos.set(System.nanoTime());

        CompletableFuture<@Nullable AbstractFrameProducer<T>> nextVideoTask = this.nextVideoTask.getAndSet(null);
        AbstractFrameProducer<T> nextVideo = nextVideoTask != null ? nextVideoTask.join() : null;

        CompletableFuture<@Nullable AudioProducer> nextAudioTask = this.nextAudioTask.getAndSet(null);
        AudioProducer nextAudio = nextAudioTask != null ? nextAudioTask.join() : null;

        AbstractFrameProducer<T> video = this.video.get();
        AudioProducer audio = this.audio.get();

        VideoArgs videoArgs = video == null ? this.newVideoArgs() : video.args;
        AudioArgs audioArgs = audio == null ? this.newAudioArgs() : audio.args;

        TimeLine timeLine = this.newTimeLine(videoArgs, audioArgs, seekTo);
        this.timeLine.set(timeLine);

        AbstractFrameProducer<T> newVideo = nextVideo != null ? nextVideo : this.buildVideo(videoArgs);
        AudioProducer newAudio = nextAudio != null ? nextAudio : this.buildAudio(audioArgs);

        boolean warmVideo = newVideo != null && !newVideo.off.get();

        if (newVideo != null && !warmVideo) newVideo.setup(seekTo);
        if (newAudio != null && newAudio.off.get()) newAudio.setup(seekTo);

        if (newVideo != null && newVideo.off.get()) {
            newVideo.shutdown();
            newVideo = null;
        }

        if (newAudio != null && newAudio.off.get()) {
            newAudio.shutdown();
            newAudio = null;
        }

        this.video.set(newVideo);
        this.audio.set(newAudio);

        if (newVideo != null) {
            int width = newVideo.args.scaledWidth(NarutoConfig.limitWidth(), NarutoConfig.limitHeight());
            int height = newVideo.args.scaledHeight(NarutoConfig.limitWidth(), NarutoConfig.limitHeight());

            this.renderer.setup(width, height, seekTo, warmVideo);

            newVideo.eager();

            AbstractFrameProducer.Frame<T> frame = newVideo.fetch(timeLine == null ? seekTo : timeLine.positionSec());
            this.renderer.update(width, height, frame == null ? null : frame.data());
        }

        this.running.set(newVideo != null || newAudio != null);
    }

    public synchronized void tick() {
        if (!this.isRunning()) this.setup(0D);

        AbstractFrameProducer<T> video = this.video.get();
        AudioProducer audio = this.audio.get();
        TimeLine timeLine = this.timeLine.get();

        if (video != null) video.eager();

        if (audio != null && video != null && timeLine != null && !timeLine.isStarted() && !this.videoDelayed.get() && System.nanoTime() - this.setupNanos.get() >= VIDEO_WARM_TIMEOUT_NANOS) {
            this.videoDelayed.set(true);
            System.err.println("[NarutoLoading] Video is not ready after " + (VIDEO_WARM_TIMEOUT_NANOS / 1_000_000_000L) + "s, letting audio start the clock.");
        }

        if (audio != null && (video == null || this.videoDelayed.get() || (timeLine != null && timeLine.isStarted()))) audio.eager();

        if (timeLine == null) return;

        if (timeLine.expired()) {
            this.synchronize(false, 0D);
            return;
        }

        if (timeLine.tick()) {
            System.err.println("[NarutoLoading] Lag spike is detected. Synchronizing...");
            this.synchronize(true, timeLine.positionSec());
            return;
        }

        if (!timeLine.isStarted()) return;

        if (video != null) {
            if (timeLine.nearEnd()) this.nextVideoTask.compareAndSet(null, CompletableFuture.supplyAsync(() -> this.buildVideo(video.args)));

            if (timeLine.due()) {
                AbstractFrameProducer.Frame<T> frame = video.fetch(timeLine.positionSec());

                if (video.consumeStarvation()) timeLine.detectLagSpike();

                this.renderer.update(video.args.scaledWidth(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), video.args.scaledHeight(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()), frame == null ? null : frame.data());
            }

            this.renderer.render();
        }

        if (audio != null && timeLine.nearEnd()) this.nextAudioTask.compareAndSet(null, CompletableFuture.supplyAsync(() -> this.buildAudio(audio.args)));
    }

    public synchronized void shutdown(boolean next) {
        this.running.set(false);
        this.videoDelayed.set(false);

        AbstractFrameProducer<T> video = this.video.getAndSet(null);
        AudioProducer audio = this.audio.getAndSet(null);

        this.timeLine.set(null);
        this.renderer.shutdown();

        if (video != null) video.shutdown();
        if (audio != null) audio.shutdown();

        if (next) {
            CompletableFuture<@Nullable AbstractFrameProducer<T>> nextVideoTask = this.nextVideoTask.getAndSet(null);
            CompletableFuture<@Nullable AudioProducer> nextAudioTask = this.nextAudioTask.getAndSet(null);

            if (nextVideoTask != null) {
                nextVideoTask.thenAccept(nextVideo -> {
                    if (nextVideo != null) nextVideo.shutdown();
                });
            }

            if (nextAudioTask != null) {
                nextAudioTask.thenAccept(nextAudio -> {
                    if (nextAudio != null) nextAudio.shutdown();
                });
            }
        }
    }

    public synchronized void synchronize(boolean next, double seekTo) {
        this.shutdown(next);
        this.setup(seekTo);
    }
}

public final class EarlySessionHandoff {
    private static volatile PlaybackSession adopted;

    private static volatile Method detachMethod;
    private static volatile boolean detachChecked;

    private EarlySessionHandoff() {
    }

    public static @Nullable PlaybackSession current() {
        PlaybackSession session = adopted;

        if (session == null) {
            Object raw = System.getProperties().remove(EarlyBridge.SESSION_PROPERTY);
            if (raw == null) return null;

            String earlyMedia = System.getProperty(EarlyBridge.MEDIA_PROPERTY);
            if (earlyMedia != null) {
                System.clearProperty(EarlyBridge.MEDIA_PROPERTY);
                NarutoContext.get().media().adopt(earlyMedia);
            }

            try {
                Class<?> type = raw.getClass();
                session = new Borrowed(raw, type.getMethod("present", boolean.class), type.getMethod("finished"), type.getMethod("close"));
            } catch (Throwable throwable) {
                System.err.println("[NarutoLoading] Unusable early video session: " + throwable.getMessage());
                return null;
            }

            adopted = session;
        }

        if (session.finished()) {
            adopted = null;
            session.close();
            return null;
        }

        return session;
    }

    public static void stop() {
        PlaybackSession session = adopted;
        adopted = null;

        if (session != null) session.close();
    }

    public static void detach() {
        Method detach = detachMethod;

        if (detach == null) {
            if (detachChecked) return;
            detachChecked = true;

            try {
                Class<?> bridge = Class.forName("me.kall.narutoloading.agent.bootstrap.EarlyBridge", true, ClassLoader.getSystemClassLoader());
                detach = bridge.getMethod("detach");
                detachMethod = detach;
            } catch (Throwable throwable) {
                return;
            }
        }

        try {
            detach.invoke(null);
        } catch (Throwable ignored) {
            // The session was already detached or closed.
        }
    }

    private record Borrowed(Object session, Method presentMethod, Method finishedMethod, Method closeMethod) implements PlaybackSession {
        @Override

            }
        }

                System.err.println("[NarutoLoading] Failed to check the early video: " + throwable.getMessage());
                return false;
            }
        }

        @Override

                this.closeMethod.invoke(this.session);

package me.kall.narutoloading.playback.produce;

import org.jetbrains.annotations.Nullable;

import java.io.File;
import java.io.IOException;
import java.io.InputStream;
import java.util.Arrays;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

public abstract class AbstractProducer {
    private static final File PRODUCER_ERROR_LOG = ErrorLog.initialize();

    public final AtomicBoolean off = new AtomicBoolean(true);

    protected final AtomicBoolean ready = new AtomicBoolean();
    private final AtomicBoolean eager = new AtomicBoolean();
    private final Object waiter = new Object();

    private final ProcessRunner runner = new ProcessRunner();
    private final AtomicReference<Runnable> warmListener = new AtomicReference<>();

    protected abstract @Nullable String[] setCommand(double seekTo);

    protected abstract void forInput(InputStream input) throws IOException, InterruptedException;

    public void setup(double seekTo) {
        String[] command = this.setCommand(seekTo);
        if (command == null || command[0] == null) return;

        this.off.set(false);
        this.onSetup(seekTo);

        this.runner.start(command, PRODUCER_ERROR_LOG, this::forInput, throwable -> {
            if (this.off.get()) return;
            System.err.println("[NarutoLoading] Exception during " + Arrays.toString(command) + " execution: " + throwable.getMessage());
        });
    }

    protected void onSetup(double seekTo) {}

    protected AbstractProducer withWarmListener(Runnable warmListener) {
        this.warmListener.set(warmListener);
        return this;
    }

    protected void notifyWarmListener() {
        Runnable warmListener = this.warmListener.get();
        if (warmListener != null) warmListener.run();
    }

    public void eager() {
        synchronized (this.waiter) {
            this.eager.set(true);
            this.waiter.notify();
        }
    }

    protected boolean awaitEager() {
        synchronized (this.waiter) {
            while (!this.eager.get() && !this.off.get()) {
                try {
                    this.waiter.wait();
                } catch (InterruptedException exception) {
                    Thread.currentThread().interrupt();
                    return false;
                }
            }
            return this.eager.get() && !this.off.get();
        }
    }

    public void shutdown() {
        this.off.set(true);

        synchronized (this.waiter) {
            this.eager.set(false);
            this.ready.set(false);
            this.waiter.notify();
        }

        this.runner.stop();
    }
}

package me.kall.narutoloading.playback.produce.audio;

import me.kall.narutoloading.media.ffmpeg.data.AudioArgs;
import me.kall.narutoloading.playback.produce.AbstractProducer;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;
import org.lwjgl.openal.*;
import org.lwjgl.system.MemoryUtil;

import java.io.IOException;
import java.io.InputStream;
import java.nio.ByteBuffer;
import java.nio.IntBuffer;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.locks.LockSupport;
import java.util.function.DoubleConsumer;

public final class AudioProducer extends AbstractProducer {
    private static final double CHUNK_SECOND = 0.1D;

    private final AtomicInteger volume;
    private final AtomicBoolean volumeChanged = new AtomicBoolean(true);

    private final String ffmpeg;

    public final AudioArgs args;

    private final int chunkSamples, preRoll;

    private final int openALFormat;

    private final long chunkNanos;

    public AudioProducer(String ffmpeg, float volume, @NotNull AudioArgs args, double bufferSeconds) {
        this.ffmpeg = ffmpeg;

        this.volume = new AtomicInteger(Float.floatToIntBits(volume));
        this.args = args;

        this.openALFormat = args.channels() == 1 ? AL11.AL_FORMAT_MONO16 : AL11.AL_FORMAT_STEREO16;

        this.chunkSamples = Math.max(1, (int) Math.round(args.sampleRate() * CHUNK_SECOND));
        this.preRoll = Math.max(1, (int) Math.round(bufferSeconds / CHUNK_SECOND));
        this.chunkNanos = (long) ((double) this.chunkSamples / args.sampleRate() * 1_000_000_000D);
    }

    public float volume() {
        return Float.intBitsToFloat(this.volume.get());
    }

    @Override
    public AudioProducer withWarmListener(Runnable warmListener) {
        super.withWarmListener(warmListener);
        return this;
    }

    public void volume(float volume) {
        if (this.volume() != volume) {
            this.volume.set(Float.floatToIntBits(volume));
            this.volumeChanged.set(true);
        }
    }

    @Override
    protected String @Nullable [] setCommand(double seekTo) {
        if (this.ffmpeg == null) return null;
        return new String[]{this.ffmpeg, "-ss", String.valueOf(seekTo + this.args.startSec()), "-i", this.args.absPath(), "-vn", "-f", "s16le", "-ac", String.valueOf(this.args.channels()), "-ar", String.valueOf(this.args.sampleRate()), "-loglevel", "error", "-"};
    }

    @Override
    protected void forInput(InputStream input) throws IOException {
        long device = 0;
        long context = 0;
        int source = 0;
        ByteBuffer direct = null;

        try {
            device = ALC11.alcOpenDevice((ByteBuffer) null);
            context = ALC11.alcCreateContext(device, (IntBuffer) null);
            EXTThreadLocalContext.alcSetThreadContext(context);
            AL.createCapabilities(ALC.createCapabilities(device));

            source = AL11.alGenSources();
            AL11.alSourcef(source, AL11.AL_GAIN, this.volume());
            this.volumeChanged.set(false);

            byte[] chunk = new byte[this.args.channels() * 2 * this.chunkSamples];
            direct = MemoryUtil.memAlloc(chunk.length);

            for (int index = 0; index < this.preRoll && !this.off.get(); index++) {
                int read = this.readFully(input, chunk);
                if (read <= 0) break;
                this.queue(source, direct, chunk, read);
            }

            if (!this.awaitEager()) return;

            this.notifyWarmListener();

            this.ready.set(true);
            AL11.alSourcePlay(source);

            boolean ended = false;

            while (!this.off.get()) {
                int processed = AL11.alGetSourcei(source, AL11.AL_BUFFERS_PROCESSED);
                while (processed-- > 0) AL11.alDeleteBuffers(AL11.alSourceUnqueueBuffers(source));

                while (!this.off.get() && AL11.alGetSourcei(source, AL11.AL_BUFFERS_QUEUED) < this.preRoll) {
                    int read = this.readFully(input, chunk);
                    if (read <= 0) {
                        ended = true;
                        break;
                    }
                    this.queue(source, direct, chunk, read);
                }

                if (ended) break;

                if (AL11.alGetSourcei(source, AL11.AL_SOURCE_STATE) != AL11.AL_PLAYING) AL11.alSourcePlay(source);
                if (this.volumeChanged.compareAndSet(true, false)) AL11.alSourcef(source, AL11.AL_GAIN, this.volume());

                LockSupport.parkNanos(this.chunkNanos);
                if (Thread.interrupted()) break;
            }

            while (!this.off.get() && AL11.alGetSourcei(source, AL11.AL_BUFFERS_QUEUED) > 0) {
                int processed = AL11.alGetSourcei(source, AL11.AL_BUFFERS_PROCESSED);
                while (processed-- > 0) AL11.alDeleteBuffers(AL11.alSourceUnqueueBuffers(source));

                LockSupport.parkNanos(this.chunkNanos);
                if (Thread.interrupted()) return;
            }
        } finally {
            if (source != 0) {
                AL11.alSourceStop(source);
                int queued = AL11.alGetSourcei(source, AL11.AL_BUFFERS_QUEUED);
                while (queued-- > 0) AL11.alDeleteBuffers(AL11.alSourceUnqueueBuffers(source));
                AL11.alDeleteSources(source);
            }

            if (context != 0) {
                EXTThreadLocalContext.alcSetThreadContext(0);
                ALC11.alcDestroyContext(context);
            }

            if (device != 0) ALC11.alcCloseDevice(device);

            if (direct != null) MemoryUtil.memFree(direct);
        }
    }

    private void queue(int source, @NotNull ByteBuffer direct, byte[] chunk, int read) {
        direct.clear();
        direct.put(chunk, 0, read).flip();

        int buffer = AL11.alGenBuffers();
        AL11.alBufferData(buffer, this.openALFormat, direct, this.args.sampleRate());
        AL11.alSourceQueueBuffers(source, buffer);
    }

    private int readFully(InputStream input, byte @NotNull [] chunk) throws IOException {
        int total = 0;

        while (total < chunk.length) {
            int read = input.read(chunk, total, chunk.length - total);
            if (read == -1) break;
            total += read;
        }

        return total - total % (this.args.channels() * 2);
    }
}

package me.kall.narutoloading.playback.produce;

import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.io.File;
import java.io.IOException;
import java.io.InputStream;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.atomic.AtomicReference;
import java.util.function.Consumer;

public final class ProcessRunner {
    private final AtomicReference<ExecutorService> executor = new AtomicReference<>();
    private final AtomicReference<Process> process = new AtomicReference<>();
    private final AtomicReference<InputStream> input = new AtomicReference<>();

    public void start(String @NotNull [] command, @Nullable File errorLog, @NotNull InputHandler handler, @NotNull Consumer<Throwable> onFailure) {
        ExecutorService executor = Executors.newSingleThreadExecutor(task -> {
            Thread thread = new Thread(task, "NarutoProducer_" + System.nanoTime());
            thread.setDaemon(true);
            return thread;
        });

        this.executor.set(executor);

        executor.submit(() -> {
            try {
                ProcessBuilder builder = new ProcessBuilder(command);
                if (errorLog != null) builder.redirectError(errorLog);

                Process process = builder.start();
                this.process.set(process);

                InputStream input = process.getInputStream();
                this.input.set(input);

                handler.handle(input);
            } catch (IOException | InterruptedException exception) {
                onFailure.accept(exception);
            }
        });
    }

    public void stop() {
        Process process = this.process.getAndSet(null);
        if (process != null) process.destroyForcibly();

        ExecutorService executor = this.executor.getAndSet(null);
        if (executor != null) executor.shutdownNow();

        InputStream input = this.input.getAndSet(null);
        if (input != null) {
            try {
                input.close();
            } catch (IOException exception) {
                System.err.println("[NarutoLoading] Failed to close input stream: " + exception.getMessage());
            }
        }
    }

    public interface InputHandler {
        void handle(@NotNull InputStream input) throws IOException, InterruptedException;
    }
}

    private final long[] costs;
    private final long budgetNanos;

    public TimeCosts(String label, int size, double budgetMillis) {
        this.label = label;
        this.size = Math.max(1, size);
        this.costs = new long[this.size];
        this.budgetNanos = budgetMillis > 0D ? (long) (budgetMillis * NANOS_PER_MILLI) : 0L;

        this.index = (this.index + 1) % this.size;
        if (this.count < this.size) this.count++;
    }

        if (n == 0) return this.label + " n=0";

        long sum = 0L;
        for (long cost : window) sum += cost;

        StringBuilder out = new StringBuilder(72);
        out.append(this.label).append(" n=").append(n);
        out.append(" avg=").append(millis(sum / (double) n)).append("ms");
        out.append(" p50=").append(millis(window[n / 2])).append("ms");
        out.append(" p95=").append(millis(window[Math.min(n - 1, (int) Math.ceil(n * 0.95D) - 1)])).append("ms");
        out.append(" max=").append(millis(window[n - 1])).append("ms");

        if (this.budgetNanos > 0L) {
            int over = 0;
            for (long cost : window) if (cost > this.budgetNanos) over++;
            out.append(" over=").append(over).append('/').append(n);
        }

        return out.toString();
    }

    private synchronized long @NotNull [] sortedWindow() {
        long[] window = new long[this.count];
        if (this.count == 0) return window;

        int start = (this.index - this.count + this.size) % this.size;
        for (int i = 0; i < this.count; i++) window[i] = this.costs[(start + i) % this.size];
        Arrays.sort(window);
        return window;
    }

    private static @NotNull String millis(double nanos) {
        return String.format(Locale.ROOT, "%.2f", nanos / NANOS_PER_MILLI);
    }
}

package me.kall.narutoloading.playback.produce.util;

import org.jetbrains.annotations.Contract;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicLong;
import java.util.function.DoubleConsumer;

public class TimeLine {
    public static final double NANO_TO_SEC = 1_000_000_000.0D;

    private final AtomicLong setupTime = new AtomicLong();
    private final AtomicLong seekNanos = new AtomicLong();
    private final AtomicBoolean started = new AtomicBoolean();
    private final long duration;
    private final double fps;

    private final AtomicLong pausedAt = new AtomicLong();
    private final AtomicBoolean paused = new AtomicBoolean();

    private final AtomicLong lastUpdate = new AtomicLong(-1);
    private final AtomicLong lastLagSpike = new AtomicLong(-1);
    private final AtomicBoolean lagSpike = new AtomicBoolean();

    private @Nullable Runnable pauseListener;
    private @Nullable DoubleConsumer resumeListener;

    private TimeLine(double duration, double fps) {
        this.duration = (long) (duration * NANO_TO_SEC);
        this.fps = fps;
    }

    @Contract("_, _ -> new")
    public static @NotNull TimeLine video(double duration, double fps) {
        return new TimeLine(duration, fps);
    }

    @Contract("_ -> new")
    public static @NotNull TimeLine audio(double duration) {
        return new TimeLine(duration, Double.POSITIVE_INFINITY);
    }

    public TimeLine withPauseListener(@NotNull Runnable pauseListener) {
        this.pauseListener = pauseListener;
        return this;
    }

    public TimeLine withResumeListener(@NotNull DoubleConsumer resumeListener) {
        this.resumeListener = resumeListener;
        return this;
    }

    public TimeLine seekTo(double at) {
        long nanos = (long) (at * NANO_TO_SEC);
        this.seekNanos.set(nanos);
        if (this.started.get()) this.setupTime.set(System.nanoTime() - nanos);
        this.lastUpdate.set(-1L);
        return this;
    }

    public void start() {
        if (!this.started.compareAndSet(false, true)) return;
        this.setupTime.set(System.nanoTime() - this.seekNanos.get());
        this.lastUpdate.set(-1L);
    }

    public boolean isStarted() {
        return this.started.get();
    }

    public void detectLagSpike() {
        this.lagSpike.set(true);
    }

    public boolean tick() {
        if (!this.started.get() || this.paused.get()) return false;
        if (!this.lagSpike.compareAndSet(true, false)) return false;

        long last = this.lastLagSpike.get();
        long now = System.nanoTime();

        if (last == -1L) {
            this.lastLagSpike.set(now);
            return false;
        }

        if (now - last <= NANO_TO_SEC * 2D) return false;

        this.lastLagSpike.set(now);
        return true;
    }

    public boolean due() {
        if (this.paused.get()) return false;

        long now = System.nanoTime();
        long last = this.lastUpdate.get();
        if (last == -1L || ((double) now - last) * this.fps >= NANO_TO_SEC) {
            this.lastUpdate.set(now);
            return true;
        }

        return false;
    }

    public boolean paused() {
        return this.paused.get();
    }

    public void pause() {
        if (!this.paused.compareAndSet(false, true)) return;

        this.pausedAt.set(System.nanoTime());
        if (this.pauseListener != null) this.pauseListener.run();
    }

    public void resume() {
        if (!this.paused.compareAndSet(true, false)) return;

        this.setupTime.addAndGet(System.nanoTime() - this.pausedAt.get());
        if (this.resumeListener != null) this.resumeListener.accept(this.positionSec());
    }

    public double positionSec() {
        return (double) this.elapsed() / NANO_TO_SEC;
    }

    public boolean nearEnd() {
        return (double) (this.duration - this.elapsed()) / NANO_TO_SEC <= 2.0D;
    }

    public boolean expired() {
        return this.started.get() && this.elapsed() >= this.duration;
    }

    private long elapsed() {
        long now = System.nanoTime();
        long origin = this.started.get() ? this.setupTime.get() : now - this.seekNanos.get();
        return now - origin;
    }
}

package me.kall.narutoloading.playback.produce.video;

import me.kall.narutoloading.media.ffmpeg.data.VideoArgs;
import me.kall.narutoloading.config.NarutoConfig;
import me.kall.narutoloading.playback.produce.AbstractProducer;
import me.kall.narutoloading.playback.produce.util.TimeCosts;
import org.jetbrains.annotations.Nullable;

import java.io.IOException;
import java.io.InputStream;
import java.nio.ByteBuffer;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.LinkedBlockingQueue;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicReference;

public abstract class AbstractFrameProducer<T> extends AbstractProducer {
    private static final long NANOS_PER_SECOND = 1_000_000_000L;

    private final String ffmpeg;

    public final VideoArgs args;

    private final boolean gpuAccel;

    private final TimeCosts onBuiltCosts;
    private final TimeCosts loopCosts;

    private long nextReportNanos;

    protected final int frameSize, bufferCapacity;
    protected final LinkedBlockingQueue<Frame<T>> frames;

    private final AtomicReference<T> lastFetched = new AtomicReference<>();

    private final AtomicLong frameIndex = new AtomicLong();

    private final AtomicBoolean waiting = new AtomicBoolean();
    private final AtomicReference<ExecutorService> waitExecutor = new AtomicReference<>();

    private final AtomicBoolean starved = new AtomicBoolean();

    public AbstractFrameProducer(String ffmpeg, VideoArgs args, boolean gpuAccel, int frameSize, double bufferSeconds) {
        this.ffmpeg = ffmpeg;

        this.args = args;
        this.gpuAccel = gpuAccel;

        this.frameSize = frameSize;
        this.bufferCapacity = (int) (this.args.fps() * bufferSeconds);
        this.frames = new LinkedBlockingQueue<>(this.bufferCapacity);
        this.nextReportNanos = System.nanoTime() + reportIntervalNanos();

        this.onBuiltCosts = new TimeCosts("onBuilt", this.bufferCapacity / 2);
        this.loopCosts = new TimeCosts("frameLoop", this.bufferCapacity / 2);
    }

    @Override
    public AbstractFrameProducer<T> withWarmListener(Runnable warmListener) {
        super.withWarmListener(warmListener);
        return this;
    }

    public @Nullable Frame<T> fetch(double positionSec) {
        if (!this.ready.get()) return null;

        this.removeLast();

        long expected = (long) (positionSec * this.args.fps());

        Frame<T> frame = this.frames.poll();

        while (frame != null && frame.index < expected) {
            this.onDead(frame.data);
            frame = this.frames.poll();
        }

        if (frame == null) {
            this.starved.set(true);
        } else {
            this.lastFetched.set(frame.data);
        }

        return frame;
    }

    public boolean consumeStarvation() {
        if (!this.starved.getAndSet(false)) return false;

        this.reportTimeCosts("starved");

        return true;
    }

    private void removeLast() {
        T lastFetched = this.lastFetched.getAndSet(null);
        if (lastFetched != null) this.onDead(lastFetched);
    }

    @Override
    protected @Nullable String[] setCommand(double seekTo) {
        if (this.ffmpeg == null) return null;

        List<String> command = new ArrayList<>();

        command.add(this.ffmpeg);
        command.add("-nostdin");

        if (this.gpuAccel) {
            command.add("-hwaccel");
            command.add("auto");
        }

        command.add("-threads");
        command.add("0");

        command.add("-ss");
        command.add(String.valueOf(seekTo + this.args.startSec()));

        command.add("-i");
        command.add(this.args.absPath());

        command.add("-map");
        command.add("0:v:0");

        command.add("-an");
        command.add("-sn"); command.add("-dn");

        command.add("-vf");
        command.add("scale=" + this.args.scaledWidth(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()) + ":" + this.args.scaledHeight(NarutoConfig.limitWidth(), NarutoConfig.limitHeight()) + ":flags=fast_bilinear,format=" + this.frameType());

        command.add("-f");
        command.add("rawvideo");

        command.add("-vcodec");
        command.add("rawvideo");

        command.add("-loglevel");
        command.add("error");

        command.add("-");

        return command.toArray(String[]::new);
    }

    protected abstract String frameType();

    @Override
    protected void forInput(InputStream input) throws IOException, InterruptedException {
        byte[] bufferArray = new byte[Math.min(this.frameSize, 64 * 1024)];

        while (!this.off.get()) {
            long loopStart = System.nanoTime();

            ByteBuffer frame = this.next();
            frame.clear();

            while (frame.hasRemaining()) {
                int read = input.read(bufferArray, 0, Math.min(bufferArray.length, frame.remaining()));
                if (read == -1) return;
                frame.put(bufferArray, 0, read);
            }

            long builtStart = System.nanoTime();
            this.onBuilt(frame.flip(), this.frameIndex.incrementAndGet());
            this.onBuiltCosts.record(System.nanoTime() - builtStart);

            this.warm();

            this.loopCosts.record(System.nanoTime() - loopStart);

            this.reportIfDue();
        }
    }

    private void reportIfDue() {
        if (!NarutoConfig.debugTimeCosts()) return;

        long now = System.nanoTime();
        if (now - this.nextReportNanos < 0L) return;

        this.nextReportNanos = now + reportIntervalNanos();
        this.reportTimeCosts("periodic");
    }

    private void reportTimeCosts(String reason) {
        System.err.println("[NarutoLoading] Time costs (" + reason + "): " + this.onBuiltCosts.describe() + " | " + this.loopCosts.describe());
        this.onBuiltCosts.reset();
        this.loopCosts.reset();
    }

    private static long reportIntervalNanos() {
        return Math.max(1, NarutoConfig.debugTimeCostsInterval()) * NANOS_PER_SECOND;
    }

    private void warm() {
        ExecutorService waitExecutor = this.waitExecutor.get();
        if (this.frames.remainingCapacity() == 0 && !this.ready.get() && !this.waiting.get() && waitExecutor != null) {
            this.waiting.set(true);
            CompletableFuture.runAsync(() -> {
                if (this.awaitEager()) {
                    this.notifyWarmListener();

                    this.ready.set(true);
                }
            }, waitExecutor).whenComplete((unused, throwable) -> {
                if (throwable != null) System.err.println("[NarutoLoading] Video warmup task failed: " + throwable.getMessage());
                waitExecutor.shutdown();
                this.waitExecutor.set(null);
            });
        }
    }

    protected abstract ByteBuffer next();

    protected abstract void onBuilt(ByteBuffer frame, long frameIndex) throws InterruptedException;
    protected abstract void onDead(T frame);

    @Override
    protected void onSetup(double seekTo) {
        this.frameIndex.set((long)(seekTo * this.args.fps()));

        ExecutorService waitExecutor = Executors.newSingleThreadExecutor(task -> {
            Thread thread = new Thread(task, "NarutoWaiter_" + System.nanoTime());
            thread.setDaemon(true);
            return thread;
        });

        this.waitExecutor.set(waitExecutor);
        this.waiting.set(false);
    }

    @Override
    public void shutdown() {
        super.shutdown();

        ExecutorService waitExecutor = this.waitExecutor.getAndSet(null);
        if (waitExecutor != null) waitExecutor.shutdown();

        for (Frame<T> frame : this.frames) {
            this.onDead(frame.data);
        }

        this.frames.clear();

        this.removeLast();
    }

    public record Frame<T>(T data, long index) {}
}

    private final Set<ByteBuffer> all = ConcurrentHashMap.newKeySet();

        for (int index = 0; index < this.bufferCapacity; index++) {
            ByteBuffer buffer = MemoryUtil.memAlloc(this.frameSize);
            this.all.add(buffer);

        }

    public void shutdown() {

        this.all.clear();

    }

        ByteBuffer buffer = this.free.poll();

        if (buffer == null) {

            this.all.add(buffer);
        }

    }

    @Override
    protected void onBuilt(ByteBuffer frame, long frameIndex) throws InterruptedException {
        this.frames.put(new Frame<>(frame, frameIndex));
    }

    @Override
    protected void onDead(ByteBuffer frame) {
        this.free.offer(frame);
    }

package me.kall.narutoloading.playback.render;

import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

import java.awt.*;
import java.awt.image.BufferedImage;
import java.awt.image.DataBufferByte;
import java.nio.ByteBuffer;

public final class Loading {
    private int width;
    private int height;

    private @Nullable ByteBuffer buffer;

    public @NotNull ByteBuffer get(int width, int height) {
        if (this.width == width && this.height == height && this.buffer != null) return this.buffer.rewind();

        BufferedImage image = new BufferedImage(width, height, BufferedImage.TYPE_BYTE_GRAY);
        genImage(width, height, image);

        int ySize = width * height;
        int uvSize = (width / 2) * (height / 2);
        int total = ySize + 2 * uvSize;

        ByteBuffer buffer = this.buffer;
        if (buffer == null || buffer.capacity() < total) buffer = ByteBuffer.allocateDirect(total);

        buffer.clear();

        byte[] gray = ((DataBufferByte) image.getRaster().getDataBuffer()).getData();
        buffer.put(gray, 0, ySize);
        for (int i = 0; i < 2 * uvSize; i++) buffer.put((byte) 128);
        buffer.flip();

        image.flush();
        this.buffer = buffer;
        this.width = width;
        this.height = height;
        return buffer;
    }

    public static void genImage(int width, int height, @NotNull BufferedImage image) {
        Graphics2D graphics = image.createGraphics();
        graphics.setColor(Color.BLACK);
        graphics.fillRect(0, 0, width, height);
        graphics.setColor(Color.WHITE);

        int fontSize = Math.max(12, height / 6);
        Font font = new Font(Font.SANS_SERIF, Font.PLAIN, fontSize);
        graphics.setFont(font);
        FontMetrics metrics = graphics.getFontMetrics();
        String text = "Loading...";
        int textWidth = metrics.stringWidth(text);
        while (textWidth > width * 0.9 && fontSize > 10) {
            fontSize--;
            font = new Font(Font.SANS_SERIF, Font.PLAIN, fontSize);
            graphics.setFont(font);
            metrics = graphics.getFontMetrics();
            textWidth = metrics.stringWidth(text);
        }

        int textHeight = metrics.getHeight();
        int x = (width - textWidth) / 2;

package me.kall.narutoloading.playback.render;

import com.mojang.blaze3d.systems.RenderSystem;
import com.mojang.blaze3d.vertex.BufferBuilder;
import com.mojang.blaze3d.vertex.BufferUploader;
import com.mojang.blaze3d.vertex.DefaultVertexFormat;
import com.mojang.blaze3d.vertex.Tesselator;
import com.mojang.blaze3d.vertex.VertexFormat;
import me.kall.narutoloading.NarutoLoading;
import me.kall.narutoloading.client.context.Graphics;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.renderer.ShaderInstance;
import net.minecraft.resources.ResourceLocation;
import org.jetbrains.annotations.Nullable;
import org.joml.Matrix4f;

import java.io.IOException;
import java.nio.ByteBuffer;

public final class ShaderBufferRenderer implements FrameRenderer<ByteBuffer> {
    private static final ResourceLocation SHADER = ResourceLocation.fromNamespaceAndPath(NarutoLoading.MOD_ID, "yuv");

    private final Loading loading = new Loading();

    private @Nullable YuvTextures planes;
    private @Nullable ShaderInstance shader;

    @Override
    public void setup(int width, int height, double seekTo, boolean ready) {
        if (this.planes != null) this.planes.close();
        this.planes = new YuvTextures(width, height);

        if (!ready) this.planes.upload(this.loading.get(width, height));

        if (this.shader == null) this.shader = loadShader();
    }

    @Override
    public void update(int width, int height, @Nullable ByteBuffer frame) {
        if (frame == null || this.planes == null) return;
        this.planes.upload(frame);
    }

    @Override
    public void render() {
        YuvTextures planes = this.planes;
        ShaderInstance shader = this.shader;
        GuiGraphics graphics = Graphics.get();

        if (planes == null || shader == null || graphics == null || planes.isEmpty()) return;

        RenderSystem.assertOnRenderThread();

        RenderSystem.setShader(() -> shader);
        RenderSystem.setShaderTexture(0, planes.texture(0));
        RenderSystem.setShaderTexture(1, planes.texture(1));
        RenderSystem.setShaderTexture(2, planes.texture(2));

        float width = graphics.guiWidth();
        float height = graphics.guiHeight();
        Matrix4f pose = graphics.pose().last().pose();

        BufferBuilder builder = Tesselator.getInstance().getBuilder();
        builder.begin(VertexFormat.Mode.QUADS, DefaultVertexFormat.POSITION_TEX);
        builder.vertex(pose, 0F, height, 0F).uv(0F, 1F).endVertex();
        builder.vertex(pose, width, height, 0F).uv(1F, 1F).endVertex();
        builder.vertex(pose, width, 0F, 0F).uv(1F, 0F).endVertex();
        builder.vertex(pose, 0F, 0F, 0F).uv(0F, 0F).endVertex();
        BufferUploader.drawWithShader(builder.end());
    }

    @Override
    public void shutdown() {
        if (this.planes != null) {
            this.planes.close();
            this.planes = null;
        }

        if (this.shader != null) {
            this.shader.close();
            this.shader = null;
        }
    }

    private static @Nullable ShaderInstance loadShader() {
        try {
            return new ShaderInstance(Minecraft.getInstance().getResourceManager(), SHADER, DefaultVertexFormat.POSITION_TEX);
        } catch (IOException exception) {
            System.err.println("[NarutoLoading] Could not load the yuv shader: " + exception.getMessage());
            return null;
        }
    }
}

package me.kall.narutoloading.playback.render;

import org.lwjgl.system.MemoryUtil;

import java.nio.ByteBuffer;
import java.nio.FloatBuffer;

import static org.lwjgl.opengl.GL32C.*;

public final class YuvPresenter implements AutoCloseable {
    private static final String VERTEX_SOURCE = """
            #version 150 core

            in vec2 Position;
            uniform float uFlipY;
            out vec2 vTex;

            void main() {
                vTex = vec2(Position.x * 0.5 + 0.5, mix(0.5 - Position.y * 0.5, 0.5 + Position.y * 0.5, uFlipY));
                gl_Position = vec4(Position, 0.0, 1.0);
            }
            """;

    private static final String FRAGMENT_SOURCE = """
            #version 150 core

            uniform sampler2D uTexY;
            uniform sampler2D uTexU;
            uniform sampler2D uTexV;
            in vec2 vTex;
            out vec4 fragColor;

            void main() {
                float y = texture(uTexY, vTex).r;
                float u = texture(uTexU, vTex).r;
                float v = texture(uTexV, vTex).r;

                float c = y - 16.0 / 255.0;
                float d = u - 128.0 / 255.0;
                float e = v - 128.0 / 255.0;

                float r = 1.164 * c + 1.596 * e;
                float g = 1.164 * c - 0.392 * d - 0.813 * e;
                float b = 1.164 * c + 2.017 * d;

                fragColor = vec4(clamp(vec3(r, g, b), 0.0, 1.0), 1.0);
            }
            """;

    private final boolean flipY;

    private int program = -1;
    private int vao;
    private int vbo;
    private YuvTextures planes;
    private int flipUniform;

    private int width;
    private int height;

    public YuvPresenter(boolean flipY) {
        this.flipY = flipY;
    }

    public void setup(int width, int height) {
        if (this.program != -1 && this.width == width && this.height == height) return;

        this.close();

        this.width = width;
        this.height = height;

        this.program = createProgram();
        this.flipUniform = glGetUniformLocation(this.program, "uFlipY");
        this.planes = new YuvTextures(width, height);

        this.vao = glGenVertexArrays();
        this.vbo = glGenBuffers();
        glBindVertexArray(this.vao);
        glBindBuffer(GL_ARRAY_BUFFER, this.vbo);

        float[] vertices = {-1F, -1F, 1F, -1F, -1F, 1F, 1F, 1F};
        FloatBuffer buffer = MemoryUtil.memAllocFloat(vertices.length);
        try {
            buffer.put(vertices).flip();
            glBufferData(GL_ARRAY_BUFFER, buffer, GL_STATIC_DRAW);
        } finally {
            MemoryUtil.memFree(buffer);
        }

        glVertexAttribPointer(0, 2, GL_FLOAT, false, 0, 0L);
        glEnableVertexAttribArray(0);
        glBindBuffer(GL_ARRAY_BUFFER, 0);
        glBindVertexArray(0);
    }

    public void upload(ByteBuffer frame) {
        if (this.planes == null || frame == null) return;
        this.planes.upload(frame);
    }

    public void present() {
        this.present(this.flipY);
    }

    public void present(boolean flip) {
        YuvTextures planes = this.planes;
        if (this.program == -1 || planes == null || planes.isEmpty()) return;

        int previousProgram = glGetInteger(GL_CURRENT_PROGRAM);
        int previousVao = glGetInteger(GL_VERTEX_ARRAY_BINDING);
        int previousActive = glGetInteger(GL_ACTIVE_TEXTURE);

        glActiveTexture(GL_TEXTURE0);
        int previousTex0 = glGetInteger(GL_TEXTURE_BINDING_2D);
        glActiveTexture(GL_TEXTURE1);
        int previousTex1 = glGetInteger(GL_TEXTURE_BINDING_2D);
        glActiveTexture(GL_TEXTURE2);
        int previousTex2 = glGetInteger(GL_TEXTURE_BINDING_2D);

        boolean blend = glIsEnabled(GL_BLEND);
        boolean depthTest = glIsEnabled(GL_DEPTH_TEST);
        boolean cullFace = glIsEnabled(GL_CULL_FACE);
        boolean scissor = glIsEnabled(GL_SCISSOR_TEST);
        boolean depthMask = glGetBoolean(GL_DEPTH_WRITEMASK);

        glDisable(GL_BLEND);
        glDisable(GL_DEPTH_TEST);
        glDisable(GL_CULL_FACE);
        glDisable(GL_SCISSOR_TEST);
        glDepthMask(false);

        glUseProgram(this.program);
        glUniform1f(this.flipUniform, flip ? 1F : 0F);

        glActiveTexture(GL_TEXTURE0);
        glBindTexture(GL_TEXTURE_2D, planes.texture(0));
        glActiveTexture(GL_TEXTURE1);
        glBindTexture(GL_TEXTURE_2D, planes.texture(1));
        glActiveTexture(GL_TEXTURE2);
        glBindTexture(GL_TEXTURE_2D, planes.texture(2));

        glBindVertexArray(this.vao);
        glDrawArrays(GL_TRIANGLE_STRIP, 0, 4);

        glActiveTexture(GL_TEXTURE0);
        glBindTexture(GL_TEXTURE_2D, previousTex0);
        glActiveTexture(GL_TEXTURE1);
        glBindTexture(GL_TEXTURE_2D, previousTex1);
        glActiveTexture(GL_TEXTURE2);
        glBindTexture(GL_TEXTURE_2D, previousTex2);
        glActiveTexture(previousActive);

        glBindVertexArray(previousVao);
        glUseProgram(previousProgram);

        glDepthMask(depthMask);
        if (cullFace) glEnable(GL_CULL_FACE);
        if (scissor) glEnable(GL_SCISSOR_TEST);
        if (depthTest) glEnable(GL_DEPTH_TEST);
        if (blend) glEnable(GL_BLEND);
    }

    @Override
    public void close() {
        if (this.planes != null) {
            this.planes.close();
            this.planes = null;
        }

        if (this.vbo != 0) {
            glDeleteBuffers(this.vbo);
            this.vbo = 0;
        }

        if (this.vao != 0) {
            glDeleteVertexArrays(this.vao);
            this.vao = 0;
        }

        if (this.program != -1) {
            glDeleteProgram(this.program);
            this.program = -1;
        }
    }

    private static int createProgram() {
        int previousProgram = glGetInteger(GL_CURRENT_PROGRAM);

        int vertex = compileShader(GL_VERTEX_SHADER, VERTEX_SOURCE);
        int fragment = compileShader(GL_FRAGMENT_SHADER, FRAGMENT_SOURCE);

        int program = glCreateProgram();
        glAttachShader(program, vertex);
        glAttachShader(program, fragment);
        glBindAttribLocation(program, 0, "Position");
        glLinkProgram(program);
        glDeleteShader(vertex);
        glDeleteShader(fragment);

        if (glGetProgrami(program, GL_LINK_STATUS) == GL_FALSE) {
            String log = glGetProgramInfoLog(program);
            glDeleteProgram(program);
            throw new IllegalStateException("YUV program link failed: " + log);
        }

        glUseProgram(program);
        glUniform1i(glGetUniformLocation(program, "uTexY"), 0);
        glUniform1i(glGetUniformLocation(program, "uTexU"), 1);
        glUniform1i(glGetUniformLocation(program, "uTexV"), 2);
        glUseProgram(previousProgram);

        return program;
    }

    private static int compileShader(int type, String source) {
        int shader = glCreateShader(type);
        glShaderSource(shader, source);
        glCompileShader(shader);

        if (glGetShaderi(shader, GL_COMPILE_STATUS) == GL_FALSE) {
            String log = glGetShaderInfoLog(shader);
            glDeleteShader(shader);
            throw new IllegalStateException("YUV shader compile failed: " + log);
        }

        return shader;
    }
}

package me.kall.narutoloading.playback.render;

import org.lwjgl.system.MemoryUtil;

import java.nio.ByteBuffer;

import static org.lwjgl.opengl.GL32C.*;

final class YuvTextures implements AutoCloseable {
    private final int width;
    private final int height;
    private final int[] textures = new int[3];
    private final int[] buffers = new int[6];

    private long frameCount;

    YuvTextures(int width, int height) {
        this.width = width;
        this.height = height;

        int previousTexture = glGetInteger(GL_TEXTURE_BINDING_2D);
        int previousBuffer = glGetInteger(GL_PIXEL_UNPACK_BUFFER_BINDING);

        glGenTextures(this.textures);
        createTexture(this.textures[0], width, height);
        createTexture(this.textures[1], width / 2, height / 2);
        createTexture(this.textures[2], width / 2, height / 2);

        int ySize = width * height;
        int uvSize = (width / 2) * (height / 2);

        glGenBuffers(this.buffers);
        for (int i = 0; i < 2; i++) {
            allocate(this.buffers[i], ySize);
            allocate(this.buffers[2 + i], uvSize);
            allocate(this.buffers[4 + i], uvSize);
        }

        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, previousBuffer);
        glBindTexture(GL_TEXTURE_2D, previousTexture);
    }

    int width() {
        return this.width;
    }

    int height() {
        return this.height;
    }

    boolean isEmpty() {
        return this.frameCount <= 0L;
    }

    int texture(int plane) {
        return this.textures[plane];
    }

    void upload(ByteBuffer frame) {
        int ySize = this.width * this.height;
        int uvSize = (this.width / 2) * (this.height / 2);

        int slot = (int) (this.frameCount & 1L);

        int previousTexture = glGetInteger(GL_TEXTURE_BINDING_2D);
        int previousBuffer = glGetInteger(GL_PIXEL_UNPACK_BUFFER_BINDING);
        int previousAlignment = glGetInteger(GL_UNPACK_ALIGNMENT);

        glPixelStorei(GL_UNPACK_ALIGNMENT, 1);

        stage(this.buffers[slot], frame, 0, ySize);
        stage(this.buffers[2 + slot], frame, ySize, uvSize);
        stage(this.buffers[4 + slot], frame, ySize + uvSize, uvSize);

        uploadTexture(this.textures[0], this.width, this.height, this.buffers[slot]);
        uploadTexture(this.textures[1], this.width / 2, this.height / 2, this.buffers[2 + slot]);
        uploadTexture(this.textures[2], this.width / 2, this.height / 2, this.buffers[4 + slot]);

        glPixelStorei(GL_UNPACK_ALIGNMENT, previousAlignment);
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, previousBuffer);
        glBindTexture(GL_TEXTURE_2D, previousTexture);

        this.frameCount++;
    }

    @Override
    public void close() {
        glDeleteTextures(this.textures);
        glDeleteBuffers(this.buffers);
        this.frameCount = 0L;
    }

    private static void createTexture(int texture, int width, int height) {
        glBindTexture(GL_TEXTURE_2D, texture);
        glTexImage2D(GL_TEXTURE_2D, 0, GL_R8, width, height, 0, GL_RED, GL_UNSIGNED_BYTE, (ByteBuffer) null);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
    }

    private static void allocate(int buffer, long size) {
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, buffer);
        glBufferData(GL_PIXEL_UNPACK_BUFFER, size, GL_STREAM_DRAW);
    }

    private static void stage(int buffer, ByteBuffer frame, int offset, int length) {
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, buffer);

        ByteBuffer mapped = glMapBufferRange(GL_PIXEL_UNPACK_BUFFER, 0L, length, GL_MAP_WRITE_BIT | GL_MAP_INVALIDATE_BUFFER_BIT | GL_MAP_UNSYNCHRONIZED_BIT);
        if (mapped == null) return;

        MemoryUtil.memCopy(MemoryUtil.memSlice(frame, offset, length), mapped);
        glUnmapBuffer(GL_PIXEL_UNPACK_BUFFER);
    }

    private static void uploadTexture(int texture, int width, int height, int buffer) {
        glBindTexture(GL_TEXTURE_2D, texture);
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, buffer);
        glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RED, GL_UNSIGNED_BYTE, 0L);
    }
}
